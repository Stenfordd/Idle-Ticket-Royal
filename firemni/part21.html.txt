const me = () => gameState.ldap;
function run(fn, args) { return new Promise((ok, bad) => { google.script.run.withSuccessHandler(ok).withFailureHandler(bad)[fn].apply(null, args || []); }); }
function onTab() { return A.isOpen() && A.view() === "duel"; }
function say(t, c) { K.msg = t; K.msgC = c || ""; K.forceDraw = true; draw(); }
function takePay(pay) {
(pay || []).forEach(p => {
if (p.why === "WIN") { if (p.amt > 0) TT.win(p.amt); log(`⚔ DUEL: you won!${p.amt ? " +" + fmt(p.amt) + " GC" : ""}`, "special"); if (window.Meta && Meta.count) Meta.count("duelWins", 1); }
else { gameState.coins += p.amt; log(`⚔ DUEL: ${p.why === "DRAW" ? "draw, stake returned" : "stake refunded"} +${fmt(p.amt)} GC`, "success"); }
});
if (pay && pay.length) { updateUI(); TT.saveSoon(); }
}
async function loadList() {
if (K.listBusy) return; K.listBusy = true;
try { const r = await run("dlList", []); K.listAt = Date.now(); if (r && r.ok) { K.list = r; takePay(r.pay); if (r.mine && !K.id) { K.id = r.mine; K.rev = -1; poll(); } } }
catch (e) { K.list = K.list || { tables: [], max: 4 }; }
finally { K.listBusy = false; }
if (onTab()) draw();
}
async function poll() {
clearTimeout(K.pollT); if (!K.id) return;
try {
const r = await run("dlPoll", [K.id, K.rev]);
if (r && r.ok) { takePay(r.pay);
if (r.gone) { K.id = ""; K.view = null; K.rev = -1; loadList(); return; }
if (r.view) apply(r.view); else if (r.same && K.view) { K.view.left = r.left; K.recvAt = Date.now(); } }
} catch (e) {}
if (K.id) K.pollT = setTimeout(poll, onTab() ? (K.view && K.view.status === "play" ? 1000 : 2000) : 4000);
}
function apply(v) {
const prev = K.view;
K.view = v; K.rev = v.rev; K.recvAt = Date.now();
if (v.status === "play" && v.role && !v.picked && K.ping !== v.half + ":" + v.rev && !onTab()) { K.ping = v.half + ":" + v.rev; log(`⚔ DUEL vs ${esc(v.names[1 - v.you])}: your move!`, "special"); }
if (onTab()) { draw(); if (prev && v.evSeq !== K.seq) { K.seq = v.evSeq; playEvents(v); } else K.seq = v.evSeq; }
}
function heroesOf(v) {
if (!v || !v.cards) return [null, null];
let a = H.fromCard(v.cards[v.names[0]]), b = v.names[1] ? H.fromCard(v.cards[v.names[1]]) : null;
if (v.mode === "equal" && a && b) { const L = Math.max(a.level, b.level); a = H.stdHero(a.cls, L); b = H.stdHero(b.cls, L); }

}
function cleanStake(x) { const n = Math.floor(Number(String(x).replace(/[^0-9.]/g, "")) || 0); return Math.max(0, Math.min((K.list && K.list.maxStake) || 1e9, n)); }

K.stake = stake; K.mode = mode; K.target = target;
if (gameState.coins < stake) { say("Not enough GC.", "var(--crimson)"); return; }
K.busy = true; draw();
if (stake > 0) TT.spend(stake);
saveGame(async () => {
try { const r = await run("dlCreate", [stake, mode, target]);
if (r && r.ok) { K.id = r.id; K.rev = -1; say(""); poll(); }
else { refund(stake); say({ already: "You already have a duel.", full: "All duel tables are busy.", stake: "Invalid stake.", noclass: "Your High Stakes character isn't saved yet - try again in a few seconds." }[r && r.reason] || "Could not create.", "var(--crimson)"); } }
catch (e) { refund(stake); say("Server error - stake returned.", "var(--crimson)"); }
finally { K.busy = false; loadList(); }
});
}
function join(id, stake) {
if (K.busy) return; if (gameState.coins < stake) { say("Not enough GC.", "var(--crimson)"); return; }
K.busy = true; draw();
if (stake > 0) TT.spend(stake);
saveGame(async () => {
try { const r = await run("dlJoin", [id]);
if (r && r.ok) { K.id = r.id; K.rev = -1; say(""); poll(); }
else { refund(stake); say({ already: "You already have a duel.", gone: "That duel is gone.", self: "That's your own challenge.", private: "That challenge is for someone else.", noclass: "Your High Stakes character isn't saved yet - try again." }[r && r.reason] || "Could not join.", "var(--crimson)"); loadList(); } }
catch (e) { refund(stake); say("Server error - stake returned.", "var(--crimson)"); }
finally { K.busy = false; }
});
}
async function leave() {
if (K.busy || !K.id) return; const v = K.view;
if (v && v.status === "play" && v.you >= 0 && !window.confirm("Flee the duel? Your opponent wins" + (v.stake ? " the pot." : "."))) return;
K.busy = true; try { await run("dlLeave", [K.id]); } catch (e) {}
K.busy = false; K.id = ""; K.view = null; K.rev = -1; clearTimeout(K.pollT); loadList();
}
/** v11.1: stejné mini-hry jako v kampani (HSQTE) - výsledek jde na server spolu s volbou */
function qteFor(v, choice, cb) {
const h = heroesOf(v)[v.you], Q = window.HSQTE;
if (!Q || !h) return cb({});
if (v.role === "atk") {
if (choice === "heavy") return Q.timing(h.zone || 0.3, q => cb({ t: q }));
if (choice === "ult") return Q.ult(h.cls, H.CLASSES[h.cls].skill, o => {
const oh = heroesOf(v)[1 - v.you], gu = v.gu && v.gu[v.you];
if (gu || !oh) return cb({ q: o.q });
Q.guess(H.reactsOf(oh).map(x => ({ k: x, icon: H.REACTS[x].icon, label: H.REACTS[x].name })), H.GUESS_FX[h.cls], g => cb(g ? { q: o.q, guess: g } : { q: o.q }));
});
if (choice === "gamble" && v.gr) {
let k = 0; const f = { h: h, R: () => Number(v.gr[k++ % v.gr.length]) || 0, sta: v.sta ? v.sta[v.you] : 100, st: {} };
return Q.gamble(f, H, (reels, g) => cb({ g: g }), (rs) => { const x = H.reelResult(rs); return x.bf ? "💀 BACKFIRE" : (x.jp ? "🎰 JACKPOT! " : "") + "×" + x.mult + " damage"; });
}
} else {
const oh = heroesOf(v)[1 - v.you], vsMage = oh && oh.cls === "engineer";
if (v.ultReady) {                                                   // v11.2: soupeř má ULTI - block / dodge se hraje 3× (jako UNLEASH)
if (choice === "block") return Q.parry3(rs => cb({ rings: rs }), "🛡 ULTIMATE is ready - parry all 3 rings" + (vsMage ? " (a Mage halves your block)" : ""));
if (choice === "dodge") return Q.dodge3(qs => cb({ dqs: qs }), "💨 ULTIMATE is ready - dodge 3 times");
}
if (choice === "block") return Q.parry({ k: "any", title: "🛡 BLOCK: hit the ring - a PERFECT parry stops a HEAVY / GAMBLE / ULTIMATE" + (vsMage ? " (only half vs a Mage)" : " completely"), easy: h.parryWin || 1 }, o => cb({ parry: o.parry }));
if (choice === "dodge") return Q.dodge(q => cb({ dq: q }));
if (choice === "channel") return Q.channel(q => cb({ cq: q }));
}
cb({});
}
function pick(choice) {

opt.h = half; send(choice, opt);
});
}

catch (e) { say("Server error - try again.", "var(--crimson)"); if (K.view) K.view.picked = false; }
finally { K.busy = false; draw(); poll(); }
}

function al(k, n) {
const v = K.view; if (!v || !v.setup || !K.al) return;
const cur = K.al, pts = v.setup.pts;
if (n < 0) cur[k] = Math.max(0, cur[k] + n);
else for (let i = 0; i < n; i++) {
if (cur.main + cur.vit + cur.luck < pts) { cur[k]++; continue; }
const from = ["main", "vit", "luck"].filter(x => x !== k && cur[x] > 0).sort((x, y) => cur[y] - cur[x])[0];
if (!from) break; cur[from]--; cur[k]++;
}
draw();
}

catch (e) { say("Server error - try again.", "var(--crimson)"); }
finally { K.busy = false; draw(); poll(); }
}
