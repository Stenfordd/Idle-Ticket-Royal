function cardHtml(c, cls, extra) {
const s = suit(c), r = rank(c), face = r >= 4 && r <= 6;
return `<div class="pr-card s${s} ${cls || ""}" ${extra || ""} title="${RANK_N[r]} of ${SUITS[s]}"><span class="rk">${RANKS[r]}</span><span class="si">${ICON[s]}</span>${face ? `<span class="fc">${RANKS[r]}</span>` : ""}<span class="big">${ICON[s]}</span><span class="rk2">${RANKS[r]}</span></div>`;
}
function playable(v, c) {
const r = rank(c), top = v.top;
if (v.pen > 0) return r === 0;
if (v.skip) return r === 7;
if (r === 5) return true;
return suit(c) === v.suit || r === rank(top);
}
function me() { return gameState.ldap; }
function visible() { const el = $("adv-view-prsi"); return !!(el && el.style.display !== "none" && el.offsetParent); }
function say(t, c) { P.msg = t; P.msgC = c || ""; render(); }
function run(fn, args) {
return new Promise((ok, bad) => { google.script.run.withSuccessHandler(ok).withFailureHandler(bad)[fn].apply(null, args || []); });
}
function takePay(pay) {
(pay || []).forEach(p => {
if (p.why === "WIN") { TT.win(p.amt); log(`♣ PRŠÍ: you won the table! +${fmt(p.amt)} GC`, "special"); if (window.Meta && Meta.count) { Meta.count("prsiWins", 1); } TT.announce("WON A PRŠÍ TABLE", "PRŠÍ", 0, p.amt); }
else { gameState.coins += p.amt; gameState.gambleNet = (gameState.gambleNet || 0) + p.amt; log(`♣ PRŠÍ: stake refunded +${fmt(p.amt)} GC`, "success"); }
});
if (pay && pay.length) { updateUI(); TT.saveSoon(); }
}
async function loadList() {
if (P.listBusy) return; P.listBusy = true;
try {
const r = await run("prsList", []);
P.lastList = Date.now();
if (r && r.ok) {
P.list = r; takePay(r.pay);
if (r.mine && !P.id) { P.id = r.mine; P.rev = -1; poll(); }
const waiting = r.tables.filter(t => t.status === "wait" && t.players.indexOf(me()) < 0).length;
glow(waiting > 0 && !visible());
}
} catch (e) { P.list = P.list || { tables: [], stakes: [], max: 2 }; }
finally { P.listBusy = false; }
render();
}
async function poll() {
clearTimeout(P.pollT);
if (!P.id) return;
try {
const r = await run("prsPoll", [P.id, P.rev]);
if (r && r.ok) {
takePay(r.pay);
if (r.gone) { P.id = ""; P.view = null; P.rev = -1; loadList(); return; }
if (r.view) apply(r.view);
else if (r.same && P.view) { P.view.left = r.left; P.recvAt = Date.now(); }
}
} catch (e) {}
if (P.id) P.pollT = setTimeout(poll, visible() ? 1200 : (P.view && P.view.status === "play" ? 2500 : 5000));
}
function apply(v) {
const prev = P.view;
P.view = v; P.rev = v.rev; P.recvAt = Date.now();
if (v.status === "play" && v.turn === me() && P.myTurnSeen !== v.rev + "") {
P.myTurnSeen = v.rev + "";
if (!visible()) { glow(true); if (!prev || prev.turn !== me()) log("♣ PRŠÍ: it's your turn!", "special"); }
}
if (prev && prev.status === "wait" && v.status === "play") log("♣ PRŠÍ: the game started - good luck!", "special");
render();
}
function glow(on) { const t = $("adv-tab-prsi"); if (t) t.classList.toggle("glow", !!on); }
async function create() {
if (P.busy) return;
const L = P.list || {}, stake = TT.cleanBet($("pr-stake").value, L.minStake || 10, Math.min(L.maxStake || 1e9, TT.maxBet ? TT.maxBet() : 1e9)), seats = Number($("pr-seats").value);
$("pr-stake").value = stake; P.lastStake = stake;
if (!(stake > 0)) { say("Enter a stake.", "var(--crimson)"); return; }
if (gameState.coins < stake) { say("Not enough GC for that stake.", "var(--crimson)"); return; }
P.busy = true; TT.spend(stake);
try {
const r = await run("prsCreate", [stake, seats]);
if (r && r.ok) { P.id = r.id; P.rev = -1; say(""); poll(); }
else { refund(stake); say({ already: "You are already at a table.", full: "Both tables are busy - wait a moment.", stake: "Invalid stake." }[r && r.reason] || "Could not open the table.", "var(--crimson)"); }
} catch (e) { refund(stake); say("Server error - stake returned.", "var(--crimson)"); }
finally { P.busy = false; loadList(); }
}
async function join(id, stake) {
if (P.busy) return;
if (gameState.coins < stake) { say("Not enough GC for that stake.", "var(--crimson)"); return; }
P.busy = true; TT.spend(stake);
try {
const r = await run("prsJoin", [id]);
if (r && r.ok) { P.id = r.id; P.rev = -1; say(""); poll(); }
else { refund(stake); say({ already: "You are already at a table.", seats: "The table is full.", gone: "That table is gone." }[r && r.reason] || "Could not join.", "var(--crimson)"); loadList(); }
} catch (e) { refund(stake); say("Server error - stake returned.", "var(--crimson)"); }
finally { P.busy = false; }
}
function refund(stake) { gameState.coins += stake; gameState.gambleNet = (gameState.gambleNet || 0) + stake; updateUI(); TT.saveSoon(); }
async function start() { if (P.busy || !P.id) return; P.busy = true; try { await run("prsStart", [P.id]); } catch (e) {} P.busy = false; poll(); }
async function leave() {
if (P.busy || !P.id) return;
const v = P.view;
if (v && v.status === "play" && v.out.indexOf(me()) < 0 && (v.fin || []).indexOf(me()) < 0 && !window.confirm("Leave the game? Your stake stays in the pot.")) return;
P.busy = true;
try { await run("prsLeave", [P.id]); } catch (e) {}
P.busy = false; P.id = ""; P.view = null; P.rev = -1; clearTimeout(P.pollT); loadList();
}
async function move(mv) {
const v = P.view; if (!v || P.busy || v.status !== "play" || v.turn !== me()) return;
P.busy = true;
if (mv.t === "play") {                                         // hned to ukázat (optimisticky)
v.mine = v.mine.filter(c => c !== mv.c); v.under = v.top; v.top = mv.c; v.suit = rank(mv.c) === 5 ? mv.suit : suit(mv.c); v.turn = "";
if (rank(mv.c) === 0) { v.pen += 2; if (v.pen >= 6 && window.Meta && Meta.count) Meta.count("prsiStack", 1); }
render();
}
try {
const r = await run("prsMove", [P.id, mv]);
if (r && r.view) apply(r.view);
if (r && !r.ok && r.reason === "card") say("That card can't be played.", "var(--crimson)");
} catch (e) { say("Connection problem - retrying...", "var(--crimson)"); P.rev = -1; }
P.busy = false; poll();
}
function play(c) {
const v = P.view; if (!v || v.turn !== me() || !playable(v, c)) return;
if (rank(c) === 5) { P.pick = c; render(); return; }
move({ t: "play", c: c });
}
function pickSuit(s) { const c = P.pick; P.pick = -1; if (c < 0) return; move({ t: "play", c: c, suit: s }); }
function draw() { move({ t: "draw" }); }

}

<b>7</b> = next player takes 2 (stack 7s: 4, 6, 8...). <b>Ace</b> = next player stands (only an ace beats it). <b>Queen</b> (Q = svršek) goes on anything and changes the suit. J = spodek.
First one out of cards wins. 30 s per turn - 3 timeouts in a row and you are out.</div>`;
}

function timeLeft(v) { return Math.max(0, (v.left || 0) - (Date.now() - P.recvAt)); }

let html = `<div class="pr-felt">`;
html += `<div class="pr-top"><span>♣ PRŠÍ · stake ${short(v.stake)}</span><span class="pot">POT ${fmt(v.pot || v.stake * v.players.length)} GC</span><button class="btn" style="padding:3px 10px;font-size:10px;" ${P.busy ? "disabled" : ""} onclick="Prsi.leave()">${v.status === "done" ? "◀ LOBBY" : "LEAVE"}</button></div>`;

html += `<div class="pr-done"><div class="big" style="font-size:16px;">WAITING FOR PLAYERS</div><div style="margin:10px 0;">${seats.join("")}</div>
<div class="pr-info" style="color:#cfe9d4;">Starts automatically when all ${v.seats} seats are taken${v.host === me() ? " - or start now with at least 2 players" : ""}. Closes in ${Math.ceil(Math.max(0, v.waitLeft - (Date.now() - P.recvAt)) / 1000)} s if nobody comes (stakes refunded).</div>
${v.host === me() ? `<button class="btn btn-gold" ${v.players.length >= 2 && !P.busy ? "" : "disabled"} onclick="Prsi.start()">▶ START NOW (${v.players.length} players)</button>` : ""}</div>`;

}

html += `<div class="pr-done"><div class="big">${v.winner === me() ? "🏆 YOU WIN!" : "🏆 " + esc(v.winner) + " WINS"}</div>
<div style="margin-top:8px;">${v.winner === me() ? `+${fmt(v.win || Math.floor(v.pot * 0.95))} GC` : "Better luck next time."}</div>

${turn ? `<div class="pr-tbar"><div style="width:${pctT}%"></div></div>` : ""}</div>`; }).join("")}</div>`;

return html + logHtml(v) + `</div>`;

function logHtml(v) { return `<div class="pr-log">${(v.log || []).slice().reverse().map(l => `<div>${esc(l.t)}</div>`).join("")}</div>`; }
function show() { glow(false); if (P.id) poll(); else loadList(); render(); }
function tick() {

else if (P.view && P.view.status === "wait") render();
else if (!P.id && Date.now() - P.lastList > 5000) loadList();
}
window.Prsi = { show: show, create: create, join: join, start: start, leave: leave, play: play, draw: draw,
suit: function (s) { if (s === -2) { P.pick = -1; render(); return; } pickSuit(s); }, refresh: loadList, _st: function (v) { P.lastStake = Math.floor(Number(v)) || 0; }, _se: function (v) { P.lastSeats = Number(v) || 4; }, _sc: function (el) { P.handScroll = el.scrollLeft; },
state: () => P, playable: playable };

setInterval(tick, 1000);
setInterval(() => { if (!P.id && !visible()) loadList(); }, 60000);   // svítící záložka, když někdo čeká u stolu
const prev = TT.isBusy; TT.isBusy = function () { return prev() || !!(P.view && P.view.status === "play" && P.view.out.indexOf(me()) < 0 && (P.view.fin || []).indexOf(me()) < 0); };
}
boot();
})();
