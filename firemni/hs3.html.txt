const ALL_ACTS = ["strike", "heavy", "gamble", "rest", "ult", "block", "dodge", "channel", "brace"];
let tut = null;
function tutApply(f) {
const s = tut.steps[tut.i]; if (!s) return;
if (f.phase !== s.ph) { if (s.ph === "react") return; }
s.setup(f);
f.disabled = {}; ALL_ACTS.forEach(a => { if (a !== s.a) f.disabled[a] = 99; });
f.demand = null; f.stunNext = false;
tutCoach();
}

}
