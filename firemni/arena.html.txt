function arenaApply(e, animate) {

if (animate && !document.hidden) {

if (e.type === "hit" || e.type === "dodge") { anim(atk, e.i === 0 ? "atk-l" : "atk-r"); if (e.type === "hit" && e.val) anim(def, "hit"); }
if (e.type === "bf") anim(atk, "hit");
if (e.type === "hit") { float((e.crit ? "CRIT " : e.jp ? "JACKPOT " : "") + fmt(e.val), dSide, e.jp ? "#ffb800" : e.crit ? "#ffb800" : "#ffffff", e.crit || e.jp); if (e.counter) float("COUNTER " + fmt(e.counter), aSide, "#ff4d6d"); if (e.outcome === "block") float("BLOCK", dSide, "#00f0ff"); }
else if (e.type === "dodge") float("DODGE", dSide, "#00f0ff", true);
else if (e.type === "bf") float("BACKFIRE", aSide, "#ff1744", true);
else if (e.type === "ult") float("+" + fmt(e.shield) + " 🛡", aSide, "#00ff88", true);
else if (e.type === "revive") float("RISE!", e.who === 0 ? "p" : "b", "#ffb800", true);
}
arenaBars();
}
function arenaStep() {
const rp = arena.replay; if (!rp) return;
clearTimeout(arena.timer);

arenaApply(rp.r.events[rp.idx++], true);
arena.timer = setTimeout(arenaStep, document.hidden ? 30 : 520);
}
function arenaSkip() { const rp = arena.replay; if (!rp) return; clearTimeout(arena.timer); while (rp.idx < rp.r.events.length) arenaApply(rp.r.events[rp.idx++], false); arenaEnd(); }
