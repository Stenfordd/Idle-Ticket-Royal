/** kolega: vlastní výbava + level, nikdy silnější než 60 % tebe */
function hero() {

const L = colLevel(), pts = H.statPoints(L);
const stats = { str: Math.round(pts * 0.4), vit: Math.round(pts * 0.4), end: pts - 2 * Math.round(pts * 0.4) };
const hh = H.buildHero({ cls: c.cls, level: L, stats: stats }, { atk: g.atk, hp: g.hp });
hh.crit = Math.min(0.55, hh.crit + (g.crit || 0));
const tb = 1 + 0.02 * trainLv();                                   // trénink z darovaných itemů: +2 % za level
hh.atk *= tb; hh.maxHp = Math.round(hh.maxHp * tb);
const me = A.hero();
hh.atk = Math.min(hh.atk, me.atk * POWER_CAP); hh.maxHp = Math.round(Math.min(hh.maxHp, me.maxHp * POWER_CAP));
return hh;
}
/** souboj kolegy = stejný engine, jen hraje počítač (a hůř trefuje parry a ulti než ty) */
function simFight(id, seed) {
const PS = H.PARRY_SKILL, US = H.ULT_SKILL, keep = [PS.perfect, PS.good, US.q, US.all];
PS.perfect = 0.15; PS.good = 0.35; US.q = 0.9; US.all = 0.3;
try { return H.simulate(hero(), C.mission(id), seed); }
finally { PS.perfect = keep[0]; PS.good = keep[1]; US.q = keep[2]; US.all = keep[3]; }
}
const wrCache = {};
function winRate(id) {
const c = st(), key = id + "|" + c.level + "|" + c.cls + "|" + JSON.stringify(c.gear) + "|" + A.HS().level + "|" + (gameState.prestige || 0);
if (wrCache[id] && wrCache[id].k === key) return wrCache[id].v;
let w = 0; for (let i = 0; i < 24; i++) if (simFight(id, 777 + i * 131).win) w++;
const v = w / 24; wrCache[id] = { k: key, v: v }; return v;
}
function reward(id) {
const gc = Math.round(C.gcFor(id, false) * 2 + (typeof calculateAutoGcRate === "function" ? calculateAutoGcRate() : 0) * (120 + 6 * id));   // + 2-7 min autogenu
const scrap = Math.round((typeof getChestScrapCost === "function" ? getChestScrapCost() : 100) * (1 + id / 20));
const xp = Math.round(C.xpFor(id, false) * 0.5);
return { gc: gc, scrap: scrap, xp: xp };
}

}
TT.saveSoon();
return rep;
}
/** v9.0: trénink za GC */
function train() {

if (trainLv() >= TRAIN_MAX) TT.holdStop = true;
after();
}

log(`🧑‍💼 ${esc(c.name)} took your old ${Gear.name(k, old)}.`, "success");
return true;
}

var a = $(from), b = $(to); if (!a || !b) return;
var t = a.textContent;
if (to !== "mh-p") {                                  // v9.0.1: od miliardy zkráceně, ať se to vejde (13.40B)
var n = Number(t.replace(/[^\d]/g, ""));
if (n >= 1e9) t = n >= 1e15 ? n.toExponential(2) : n >= 1e12 ? (n / 1e12).toFixed(2) + "T" : (n / 1e9).toFixed(2) + "B";
}
if (cache[to] !== t) { cache[to] = t; b.textContent = t; }
}
