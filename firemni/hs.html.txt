const RED_TAPE_MAX = 6;
function pickDemand(f) {
const m = f.e.mech;
if (m === "cobol") { f.demand = f.turn > 0 && f.R() < 0.3 && f.intent.k !== "heavy" ? "block" : null; return; }   // "Do you want a bryndza?" - odpověď BLOCK
if (m === "louis") { if (!(f.turn > 0 && f.R() < 0.3)) { f.demand = null; return; } }                              // SUPREME DECREE
else if (m !== "compliance") { f.demand = null; return; }
const pool = FORMS.filter(a => a !== f.demand && (a === "block" || (ACTIONS[a].sta <= f.sta && !(f.cd[a] > 0))));
f.demand = pool.length ? pool[Math.floor(f.R() * pool.length)] : "block";
}
function redTapeAtk(f) { return 1 - 0.08 * f.redTape; }
function allowed(f, a) {
const A = ACTIONS[a]; if (!A) return false;
if (f.disabled[a] > 0) return false;
if (a === "skill") return f.focus >= 100;
if (f.cd[a] > 0) return false;
return f.sta >= A.sta;
}
/** Break meter: zásahy plní, na 100 je nepřítel STAGGERED (přijde o tah) a další rána dá 1,5×. */
const BREAK = { quick: 7, strike: 15, power: 22, gamble: 9, skill: 30, parry: 40, perfect: 15 };
function addBreak(f, n) {
if (f.over || f.intent.k === "stun") return;
n *= 1 + ((f.h.perks || {}).breakBonus || 0);
const res = f.m.kind === "boss" || f.m.kind === "secret" || f.m.kind === "world" ? 0.65 : f.m.kind === "mini" || f.m.kind === "elite" ? 0.85 : 1;
f.brk = Math.min(100, f.brk + n * res * (f.intent.k === "charge" || f.intent.k === "unleash" ? 1.6 : 1));
}
function hurtEnemy(f, dmg, ignoreDef) {
const P = f.h.perks || {};
if (f.vuln) { dmg *= 1.5; f.vuln = false; }
if (f.firstHit) { dmg *= 2; f.firstHit = false; }
if (f.edefend && !ignoreDef) dmg *= 0.4;
if (f.e.mech === "great_i" && f.form === "l" && !ignoreDef) dmg *= 0.6;
if (f.intent && f.intent.k === "siesta") dmg *= 1.5;
dmg = Math.max(0, Math.round(dmg));
const s = Math.min(f.eshield, dmg); f.eshield -= s;
f.ehp -= dmg - s;
if (P.vamp && dmg > 0) f.hp = Math.min(f.h.maxHp, f.hp + Math.round(dmg * P.vamp));
if (dmg > f.st.maxHit) f.st.maxHit = dmg;
return dmg;
}
function hurtHero(f, dmg, src) {
dmg = Math.round(dmg);
const s = Math.min(f.shield, dmg); f.shield -= s;
f.hp -= dmg - s;
f.focus = Math.min(100, f.focus + 10);
const P = f.h.perks || {};
if (P.thorns && src !== "backfire" && dmg - s > 0) f.ehp -= Math.round((dmg - s) * P.thorns);
if (f.hp <= 0 && f.h.cls === "sysadmin" && !f.revived) { f.revived = true; f.hp = Math.round(f.h.maxHp * 0.1); ev(f, "revive", { val: f.hp }); }
if (f.hp <= 0 && P.duck && !f.duckUsed) { f.duckUsed = true; f.hp = 1; ev(f, "revive", { val: 1, src: "duck" }); }
ev(f, "hhit", { val: dmg, absorbed: s, src: src });
return dmg;
}
function atkMult(f) { return f.h.atk * (1 + f.coin) * (f.hotfix ? 2 : 1) * redTapeAtk(f) * (f.payroll > 0 ? 0.75 : 1); }
function gainSta(f, n) { f.sta = Math.min(f.h.staMax, f.sta + n + (f.h.staGain || 0)); }
function critRoll(f) { const c = f.R() < f.h.crit; if (c) f.st.crits++; return c; }
function spinReels(f) {
const L = f.h.luckReel || 0;
const w = REELS.map(([s, x]) => [s, s === "💀" ? Math.max(4, x - L * 0.3) : s === "7️⃣" ? x + L * 0.15 : s === "⭐" ? x + L * 0.1 : x]);
const sum = w.reduce((a, x) => a + x[1], 0);
const one = () => { let r = f.R() * sum; for (const x of w) { if (r < x[1]) return x[0]; r -= x[1]; } return "🍒"; };
return [one(), one(), one()];
}
/** v8.7: každý symbol má vlastní efekt, efekty se sčítají. Tři stejné = velký bonus.
*  🍒 heal · 🔔 BREAK · ⭐ focus · 7️⃣ damage · 💀 backfire (zraní tebe) */
const SYM = {
"🍒": { m: 0.2, heal: 0.04 },
"🔔": { m: 0.3, brk: 12 },
"⭐": { m: 0.3, foc: 14 },
"7️⃣": { m: 0.75 },
"💀": { m: 0, self: 0.25 }
};
const RESPIN_STA = 10;
function reelResult(r, opt) {
opt = opt || {};
const c = {}; r.forEach(s => { c[s] = (c[s] || 0) + 1; });
if (opt.wildCherry && c["🍒"]) {                                     // Office Plant: Lucky Cherry - 🍒 je wild (doplní nejčastější symbol)
const best = Object.keys(c).filter(s => s !== "🍒" && s !== "💀").sort((a, b) => c[b] - c[a] || (SYM[b].m - SYM[a].m))[0];
if (best) { c[best] += c["🍒"]; delete c["🍒"]; }
}
if (c["7️⃣"] === 3) return { mult: 6, txt: "JACKPOT", jp: true, heal: 0, brk: 30, foc: 20, self: 0 };
if (c["💀"] === 3) return { mult: 0, txt: "BACKFIRE", bf: true, heal: 0, brk: 0, foc: 0, self: 0.9 };
if (c["⭐"] === 3) return { mult: 3.5, txt: "STARS", heal: 0.15, brk: 15, foc: 60, self: 0 };
if (c["🔔"] === 3) return { mult: 3, txt: "BELLS", heal: 0, brk: 70, foc: 10, self: 0 };
if (c["🍒"] === 3) return { mult: 2.2, txt: "CHERRIES", heal: 0.25, brk: 10, foc: 10, self: 0 };
let mult = 0, heal = 0, brk = 0, foc = 0, self = 0;
Object.keys(c).forEach(s => { const x = SYM[s], n = c[s]; mult += x.m * n; heal += (x.heal || 0) * n; brk += (x.brk || 0) * n; foc += (x.foc || 0) * n; self += (x.self || 0) * n; });
const pair = Object.keys(c).find(s => c[s] === 2 && s !== "💀");
if (pair) mult += 0.4;
const txt = c["💀"] === 2 ? "two skulls" : pair ? "pair " + pair + pair : "mixed";
return { mult: Math.round(mult * 100) / 100, txt: txt, heal: heal, brk: brk, foc: foc, self: self };
}
/** GAMBLE krok 1: zatočí válce (tah ještě neproběhl). */
function gambleRoll(f) { f.gReels = spinReels(f); f.gRespun = false; return f.gReels.slice(); }
/** GAMBLE krok 2 (volitelný): podržené válce zůstanou, ostatní se přetočí. Stojí RESPIN_STA staminy, jen jednou. */
function canRespin(f) { return !!f.gReels && !f.gRespun && f.sta >= ACTIONS.gamble.sta + RESPIN_STA; }
function gambleRespin(f, hold) {
if (!canRespin(f)) return null;
const fresh = spinReels(f);
f.gReels = f.gReels.map((s, i) => hold && hold[i] ? s : fresh[i]);
f.gRespun = true; f.sta -= RESPIN_STA; f.st.respins = (f.st.respins || 0) + 1;
return f.gReels.slice();
}
/** simulace / auto: které válce podržet */
function autoHold(r) {
const c = {}; r.forEach(s => { c[s] = (c[s] || 0) + 1; });
const keep = Object.keys(c).find(s => c[s] >= 2 && s !== "💀") || (c["7️⃣"] ? "7️⃣" : null);
return r.map(s => s === keep);
}

function autoPolicy(f, R) {

if (k === "withdraw") return allowed(f, "dodge") ? "dodge" : "block";
if (allowed(f, "skill")) return "skill";
if (k === "defend" || k === "flex") return allowed(f, "quick") ? "quick" : "block";
if (k === "siesta" && allowed(f, "power")) return "power";
if (allowed(f, "power") && (f.sta >= 60 || R() < 0.5)) return "power";
if (allowed(f, "strike") && f.sta >= 45) return "strike";
if (allowed(f, "gamble") && f.sta >= 45 && R() < 0.2) return "gamble";
if (allowed(f, "quick")) return "quick";

}
