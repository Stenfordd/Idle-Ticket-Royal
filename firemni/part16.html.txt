<script>
/* HIGH STAKES - ARÉNA (PvP proti uloženým postavám, počítá server) + karta hráče z leaderboardu */
(function () {
"use strict";
const A = window.HS_API, H = window.HSEngine, E = window.DungeonEngine;
const $ = (id) => document.getElementById(id);
const fmt = A.fmt, fmtT = A.fmtT, anim = A.anim, float = A.float, sprite = A.sprite, playerSprite = A.playerSprite;
const STANCE_TXT = { safe: ["SAFE", "careful AI"], normal: ["NORMAL", "balanced AI"], allin: ["ALL-IN", "gambler AI"] };
const arena = { data: null, loading: false, err: "", atkStance: "normal", replay: null, timer: 0, busy: false };
const CLS_IDX = { sysadmin: 0, engineer: 1, cyber: 2 };
function arenaLoad() {
if (arena.loading) return;
arena.loading = true; arena.err = "";
google.script.run.withSuccessHandler(r => {
arena.loading = false;
if (r && r.ok) arena.data = r; else arena.err = "Could not load the arena.";
if (A.isOpen() && A.view() === "arena" && !arena.replay) $("dg-inner").innerHTML = arenaHtml();
}).withFailureHandler(e => { arena.loading = false; arena.err = String(e && e.message || e); if (A.isOpen() && A.view() === "arena") $("dg-inner").innerHTML = arenaHtml(); }).arenaList();
}
function stanceBtns(cur, fn) {
return ["safe", "normal", "allin"].map(s => `<button class="btn ${s === "allin" ? "btn-crimson" : (s === "safe" ? "btn-cyan" : "")} ${cur === s ? "sel" : ""}" onclick="Dungeon.${fn}('${s}')">${STANCE_TXT[s][0]}<small>${STANCE_TXT[s][1]}</small></button>`).join("");
}
function ago(ts) { const m = Math.max(0, Math.floor((Date.now() - ts) / 60000)); return m < 60 ? m + "m" : (m < 1440 ? Math.floor(m / 60) + "h" : Math.floor(m / 1440) + "d"); }
function arenaHtml() {
const d = A.HS(), a = arena.data;
if (arena.replay) return arenaReplayHtml();
const me = a ? a.me : null;
const myPow = E.power(A.player());
let opp = "";
if (!a) opp = `<div class="dg-dim">${arena.err ? "⚠ " + escapeHtml(arena.err) : "Loading opponents..."}</div>`;
else if (!a.opponents.length) opp = `<div class="dg-dim">No opponents yet. Your friends need a HIGH STAKES class first.</div>`;
else opp = a.opponents.map(o => {
const ratio = o.power / Math.max(1, myPow);
const tag = ratio > 1.25 ? `<span style="color:var(--crimson);">STRONGER</span>` : ratio < 0.8 ? `<span style="color:var(--emerald);">WEAKER</span>` : `<span style="color:var(--amber);">EVEN</span>`;
const cd = o.cooldown > 0;
const can = me && me.left > 0 && !cd && !arena.busy;
return `<div class="dg-art"><div style="width:44px;flex:none;">${sprite(CLS_IDX[o.cls] || 0, "p").replace('class="dg-sprite"', 'class="dg-sprite" style="width:40px;height:40px;margin:0;"')}</div>
<div class="t">${typeof presenceDot === "function" ? presenceDot(o) : ""}<b>${escapeHtml(o.nick)}</b> · ${E.CLASSES[o.cls] ? E.CLASSES[o.cls].icon : ""} LV ${o.level} · 🏅 <b style="color:var(--amber);">${o.honor}</b>
<div class="dg-dim">${escapeHtml(o.title || "")} · power ${fmt(o.power)} ${tag} · missions ${o.best}/51 · ${o.wins}W/${o.losses}L</div></div>
<button class="btn ${can ? "btn-crimson" : ""}" ${can ? "" : "disabled"} onclick="Dungeon.arenaAttack('${escapeHtml(o.nick).replace(/'/g, "")}')">${cd ? "⏳ " + fmtT(o.cooldown) : "⚔ ATTACK"}</button></div>`;
}).join("");
const log = a && a.log.length ? a.log.map(l => {
const iAtt = l.attacker === gameState.ldap, won = l.winner === gameState.ldap;
return `<div class="dg-dim"><span style="color:${won ? "var(--emerald)" : "var(--crimson)"};">${won ? "WIN" : "LOSS"}</span> · ${iAtt ? "you attacked <b>" + escapeHtml(l.defender) + "</b>" : "<b>" + escapeHtml(l.attacker) + "</b> attacked you"} · ${l.delta >= 0 ? "+" : ""}${l.delta} honor · ${ago(l.time)} ago</div>`;
}).join("") : `<div class="dg-dim">No arena fights yet.</div>`;
return `<div class="dg-grid2">
<div class="dg-card"><div class="dg-h"><span>🏟 YOUR ARENA CARD</span><button class="btn" style="padding:3px 8px;font-size:10px;" onclick="Dungeon.arenaRefresh()">↻</button></div>
<div class="dg-dim">Honor <b style="font-size:14px;">${me ? me.honor : "-"}</b> · ${me ? me.wins + "W / " + me.losses + "L" : ""} · attacks left today <b>${me ? me.left + " / " + me.perDay : "-"}</b> · your power <b>${fmt(myPow)}</b></div>
<div class="dg-h" style="margin-top:10px;">ATTACK STANCE</div>
<div class="dg-stances" style="margin:0;">${stanceBtns(arena.atkStance, "arenaAtkStance")}</div>
<div class="dg-h" style="margin-top:10px;">DEFENSE STANCE <span class="dg-dim" style="font-family:inherit;">(used when others attack you)</span></div>
<div class="dg-stances" style="margin:0;">${stanceBtns(d.arenaStance, "arenaDefStance")}</div>
<div class="dg-dim" style="margin-top:8px;">Fights are simulated by the server with your <b>saved</b> gear, HIGH STAKES level and class - your friends fight your build even when you are offline.
Win = honor + GC. Same opponent once per 20 min. Honor works like chess Elo: beating stronger players gives more.</div></div>
<div class="dg-card"><div class="dg-h">RECENT FIGHTS</div>${log}</div>
</div>
<div class="dg-card"><div class="dg-h"><span>OPPONENTS</span><span class="dg-dim">sorted by honor</span></div>${opp}</div>`;
}
function arenaAtkStance(s) { arena.atkStance = s; $("dg-inner").innerHTML = arenaHtml(); }
function arenaDefStance(s) { A.HS().arenaStance = s; TT.saveSoon(); $("dg-inner").innerHTML = arenaHtml(); }
function arenaAttack(nick) {
if (arena.busy) return;
arena.busy = true;
$("dg-inner").innerHTML = arenaHtml();
const go = () => google.script.run.withSuccessHandler(r => {
arena.busy = false;
if (!r || !r.ok) {
const why = { limit: "No attacks left today.", cooldown: "You fought this player recently.", target: "Opponent not available.", missing: "Save not found - try again in a few seconds.", noclass: "Pick a HIGH STAKES class first.", busy: "Server busy, try again." }[r && r.reason] || "Fight failed.";
arena.err = why; log("ARENA: " + why, "warning"); arenaLoad(); return;
}
gameState.coins += r.reward;              // odměna hned (i kdyby se replay nedokoukal)
TT.saveSoon(); updateUI();
log(`🏟 ARENA ${r.win ? "WIN" : "LOSS"} vs ${escapeHtml(r.opp.nick)}: ${r.delta >= 0 ? "+" : ""}${r.delta} honor (now ${r.honor}), +${fmt(r.reward)} GC.`, r.win ? "special" : "error");
arena.replay = { r: r, idx: 0, hp: [r.max[0], r.max[1]], sh: [0, 0], lines: [] };
$("dg-inner").innerHTML = arenaReplayHtml();
arenaStep();
}).withFailureHandler(e => { arena.busy = false; arena.err = String(e && e.message || e); arenaLoad(); }).arenaFight(nick, arena.atkStance);
saveGame(() => go());                      // nejdřív uložit aktuální build, server bojuje s uloženým
}
function arenaReplayHtml() {
const rp = arena.replay, r = rp.r, d = A.HS();
return `<div class="dg-arena" id="dg-arena">
<div class="dg-side"><div class="dg-name">${escapeHtml(gameState.ldap)}</div><div class="dg-sub">${E.CLASSES[d.cls].name} · ${STANCE_TXT[arena.atkStance][0]}</div>
<div id="ar-a">${playerSprite()}</div>
<div class="dg-bar"><div class="hp-p" id="ar-ahp" style="width:100%"></div><div class="sh" id="ar-ash" style="width:0%"></div></div><div class="dg-hpt" id="ar-ahpt"></div></div>
<div class="dg-vs">VS</div>
<div class="dg-side"><div class="dg-name" style="color:var(--crimson);">${escapeHtml(r.opp.nick)}</div><div class="dg-sub">${E.CLASSES[r.opp.cls] ? E.CLASSES[r.opp.cls].name : ""} LV ${r.opp.level} · ${STANCE_TXT[r.defStance][0]}</div>
<div id="ar-b">${sprite(CLS_IDX[r.opp.cls] || 0, "p")}</div>
<div class="dg-bar"><div class="hp-b" id="ar-bhp" style="width:100%"></div><div class="sh" id="ar-bsh" style="width:0%"></div></div><div class="dg-hpt" id="ar-bhpt"></div></div>
</div>
<div class="dg-mech" id="ar-res">${r.first === 0 ? "You strike first!" : escapeHtml(r.opp.nick) + " strikes first!"}</div>
<div class="dg-row"><button class="btn" style="flex:1;" id="ar-skip" onclick="Dungeon.arenaSkip()">SKIP ⏭</button>
<button class="btn btn-gold" style="flex:1;" id="ar-back" disabled onclick="Dungeon.arenaBack()">BACK TO ARENA</button></div>
<div class="dg-card" style="margin-top:10px;"><div class="dg-h">FIGHT LOG</div><div class="dg-log" id="ar-log"></div></div>`;
}
function arenaBars() {
const rp = arena.replay; if (!rp) return;
const m = rp.r.max;
const set = (id, w) => { const el = $(id); if (el) el.style.width = Math.max(0, Math.min(100, w)) + "%"; };
set("ar-ahp", rp.hp[0] / m[0] * 100); set("ar-bhp", rp.hp[1] / m[1] * 100);
set("ar-ash", rp.sh[0] / m[0] * 100); set("ar-bsh", rp.sh[1] / m[1] * 100);
const a = $("ar-ahpt"); if (a) a.textContent = fmt(rp.hp[0]) + " / " + fmt(m[0]);
const b = $("ar-bhpt"); if (b) b.textContent = fmt(rp.hp[1]) + " / " + fmt(m[1]);
}
function arenaLine(e) {
const r = arena.replay.r, nm = (i) => i === 0 ? "You" : escapeHtml(r.opp.nick), T = `<span style="color:#556;">T${e.t}</span> `;
if (e.type === "hit") return [T + `${nm(e.i)} ${e.kind === "jp" ? "🎰 <b>JACKPOT</b>" : "hit"} <b>${fmt(e.val)}</b> (${e.mult}×)${e.rev ? " - 🔌 POWER CYCLE!" : ""}`, e.i === 0 ? (e.kind === "jp" ? "y" : "g") : "r"];
if (e.type === "bf") return [T + `💥 ${nm(e.i)} BACKFIRED for ${fmt(e.val)}`, e.i === 0 ? "r" : "g"];
if (e.type === "dodge") return [T + `🌀 ${nm(1 - e.i)} dodged`, e.i === 0 ? "r" : "c"];
if (e.type === "reflect") return [T + `${nm(e.i)} reflected ${fmt(e.val)}`, "c"];
if (e.type === "heal") return [T + `${nm(e.i)} healed ${fmt(e.val)}`, "g"];
if (e.type === "dot") return [T + `Zero-day payload ticks ${fmt(e.val)}`, "c"];
return [T + e.type, ""];
}
function arenaApply(e, animate) {
const rp = arena.replay;
rp.hp = e.hp.slice(); rp.sh = e.sh.slice();
const ln = arenaLine(e); rp.lines.unshift(ln);
const el = $("ar-log"); if (el) { const div = document.createElement("div"); div.className = ln[1]; div.innerHTML = ln[0]; el.prepend(div); }
if (animate && !document.hidden) {
const atk = e.i === 0 ? "ar-a" : "ar-b", def = e.i === 0 ? "ar-b" : "ar-a";
if (e.type === "hit" || e.type === "dodge") { anim(atk, e.i === 0 ? "atk-l" : "atk-r"); if (e.type === "hit") anim(def, "hit"); }
if (e.type === "bf") anim(atk, "hit");
const side = (e.type === "hit" || e.type === "dot") ? (e.i === 0 ? "b" : "p") : (e.type === "bf" || e.type === "heal") ? (e.i === 0 ? "p" : "b") : (e.type === "dodge" ? (e.i === 0 ? "b" : "p") : (e.i === 0 ? "b" : "p"));
const txt = e.type === "hit" ? (e.kind === "jp" ? "JACKPOT " + fmt(e.val) : fmt(e.val)) : e.type === "bf" ? "BACKFIRE" : e.type === "dodge" ? "DODGE" : e.type === "heal" ? "+" + fmt(e.val) : fmt(e.val || 0);
float(txt, side, e.type === "hit" ? (e.kind === "jp" ? "#ffb800" : "#ffffff") : e.type === "bf" ? "#ff1744" : e.type === "dodge" ? "#00f0ff" : "#00ff88", e.kind === "jp");
}
arenaBars();
}
function arenaStep() {
const rp = arena.replay; if (!rp) return;
clearTimeout(arena.timer);
if (rp.idx >= rp.r.events.length) { arenaEnd(); return; }
arenaApply(rp.r.events[rp.idx++], true);
arena.timer = setTimeout(arenaStep, document.hidden ? 30 : 520);
}
function arenaSkip() { const rp = arena.replay; if (!rp) return; clearTimeout(arena.timer); while (rp.idx < rp.r.events.length) arenaApply(rp.r.events[rp.idx++], false); arenaEnd(); }
function arenaEnd() {
const rp = arena.replay, r = rp.r;
rp.done = true;
const res = $("ar-res");
if (res) res.innerHTML = `<span style="font:900 18px Orbitron,monospace;color:${r.win ? "var(--emerald)" : "var(--crimson)"};">${r.win ? "🏆 VICTORY" : "☠ DEFEAT"}</span><br>${r.delta >= 0 ? "+" : ""}${r.delta} honor (now ${r.honor}) · +${fmt(r.reward)} GC · ${r.left} attacks left today`;
const loser = document.querySelector(r.win ? "#ar-b .dg-sprite" : "#ar-a .dg-sprite"); if (loser) loser.classList.add("dead");
const b = $("ar-back"); if (b) b.disabled = false;
const s = $("ar-skip"); if (s) s.disabled = true;
}
function arenaRestore() {
$("dg-inner").innerHTML = arenaReplayHtml();
const rp = arena.replay;
const el = $("ar-log"); if (el) el.innerHTML = rp.lines.slice().reverse().map(l => `<div class="${l[1]}">${l[0]}</div>`).join("");
arenaBars();
if (rp.done) arenaEnd();
}
function arenaBack() { clearTimeout(arena.timer); arena.replay = null; arena.data = null; A.render(); }
function showProfile(u) {
if (!u) return;
const dg = u.dungeon || {};
const cls = dg.cls && H.CLASSES[dg.cls];
const old = $("dg-profile"); if (old) old.remove();
const el = document.createElement("div");
el.id = "dg-profile"; el.className = "tt-overlay open"; el.style.zIndex = 10000;
el.onclick = (e) => { if (e.target === el) el.remove(); };
el.innerHTML = `<div class="tt-modal" style="border-color:var(--cyan);">
<h2 style="color:var(--cyan);">${escapeHtml(u.ldap)} <span style="font:400 12px sans-serif;color:var(--text-dim);letter-spacing:0;">${typeof presenceText === "function" ? presenceText(u) : ""}</span></h2>
<div style="display:flex;gap:12px;align-items:center;">
${cls ? sprite(["sysadmin", "engineer", "cyber"].indexOf(dg.cls), "p") : '<div style="font-size:48px;">👤</div>'}
<div class="tt-body" style="min-height:0;font-size:13px;">
${cls ? `🎰 <b>${escapeHtml(dg.title || cls.name)}</b> · ${cls.name} LV ${dg.level || 1}<br>` : "Not in HIGH STAKES yet<br>"}
Missions cleared: <b>${dg.best || 0}/51</b> · Wins ${dg.wins || 0} / Losses ${dg.losses || 0} · 🏅 Arena honor <b>${dg.honor || 1000}</b><br>
Prestige <b>${u.prestige || 0}</b> · Chest LV <b>${u.chestLvl || 1}</b> · Peak CPS <b>${Number(u.peakCps || 0).toFixed(1)}</b><br>
Score <b>${fmt(u.score || 0)}</b> · Gambling <b style="color:${(u.gambleNet || 0) >= 0 ? "var(--emerald)" : "var(--crimson)"};">${(u.gambleNet || 0) >= 0 ? "+" : ""}${fmt(u.gambleNet || 0)}</b> GC
${u.gear ? `<br>Gear: ${u.gear}` : ""}
${window.Meta ? Meta.badgesHtml(u.ach) : ""}
</div></div>
<div style="text-align:right;margin-top:10px;"><button class="btn" onclick="this.closest('.tt-overlay').remove()">CLOSE</button></div></div>`;
document.body.appendChild(el);
}
A.registerMode("arena", { label: "🏟 ARENA", order: 90,
render: function (box) { box.innerHTML = arenaHtml(); if (!arena.replay) arenaLoad(); else arenaRestore(); },
busy: function () { return arena.busy || !!(arena.replay && !arena.replay.done); } });
Object.assign(window.Dungeon, { arenaAttack: arenaAttack, arenaAtkStance: arenaAtkStance, arenaDefStance: arenaDefStance, arenaSkip: arenaSkip, arenaBack: arenaBack,
arenaRefresh: function () { arena.data = null; A.render(); }, showProfile: showProfile });
})();
</script>
