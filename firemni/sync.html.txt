}

function nv(v, d) { v = Number(v); return isFinite(v) ? v : d; }
function cleanEnh(e) { return typeof e === "string" ? e.replace(/[^cam]/g, "").slice(0, 15) : ""; }
function migrateInventory(arr) {
const out = [];
(Array.isArray(arr) ? arr : []).forEach(o => {
if (!o || SLOTS.indexOf(o.slot) < 0) return;
const tier = Math.floor(nv(o.tier, 0));
if (tier < 1 || tier > 6) return;
const lvl = Math.max(0, Math.min(MERGE_MAX_LVL, Math.floor(nv(o.lvl !== undefined ? o.lvl : o.mergeLevel, 0))));
const n = o.n !== undefined ? Math.max(1, Math.floor(nv(o.n, 1))) : 1;
const e = cleanEnh(o.e);
const ex = out.find(s => s.slot === o.slot && s.tier === tier && s.lvl === lvl && (s.e || "") === e);
if (ex) ex.n += n; else { const it = { slot: o.slot, tier: tier, lvl: lvl, n: n }; if (e) it.e = e; out.push(it); }
});
return out;
}
function migrateLoadout(lo) {
const out = { HEAD: null, RIG: null, TOOL: null, AUGMENT: null, OS: null, RELIC: null };
if (!lo) return out;
SLOTS.forEach(s => {
const o = lo[s];
if (!o) return;
const tier = Math.floor(nv(o.tier, 0));
if (tier < 1 || tier > 6) return;
out[s] = { slot: s, tier: tier, lvl: Math.max(0, Math.min(MERGE_MAX_LVL, Math.floor(nv(o.lvl !== undefined ? o.lvl : o.mergeLevel, 0)))) };
const e = cleanEnh(o.e); if (e) out[s].e = e;
});
return out;
}

g.ldap = l.ldap || g.ldap;
saveRev = Math.max(0, Math.floor(nv(l.rev, 0)));
g.season = Number(l.season) || g.season;
g.tickets = Math.max(0, nv(l.tickets, 0));
g.coins = l.coins !== undefined ? Math.max(0, nv(l.coins, 500)) : 500;
g.scrap = l.scrap !== undefined ? Math.max(0, nv(l.scrap, 250)) : 250;
g.prestige = Math.max(0, Math.floor(nv(l.prestige, 0)));
g.chestLevel = Math.max(1, Math.floor(nv(l.chestLevel, 1)));
g.maxChestLevel = Math.max(g.chestLevel, Math.floor(nv(l.maxChestLevel, 1)));
g.peakCps = Math.max(0, nv(l.peakCps, 0));
g.gambleNet = nv(l.gambleNet, 0);
if (l.lifetimeTickets !== undefined && nv(l.lifetimeTickets, 0) > 0) {
g.lifetimeTickets = nv(l.lifetimeTickets, 0);
} else {
let est = g.tickets;
for (let p = 0; p < g.prestige; p++) est += getPrestigeCost(p);
g.lifetimeTickets = est;
}
if (g.lifetimeTickets < g.tickets) g.lifetimeTickets = g.tickets;
g.loadout = migrateLoadout(l.loadout);
g.inventory = migrateInventory(l.inventory);
g.autosellTier = l.autosellTier !== undefined ? Math.max(0, Math.min(4, Math.floor(nv(l.autosellTier, 0)))) : (l.autosellGrey ? 1 : 0);
g.focusAutogenActive = !!l.focusAutogenActive;
g.focusAutogenRemaining = Math.max(0, nv(l.focusAutogenRemaining, 0));
g.tutorialDone = !!l.tutorialDone;
g.slotTokens = Math.max(0, Math.floor(nv(l.slotTokens, 0)));
g.tokBuy = Math.max(0, Math.floor(nv(l.tokBuy, 0)));
if (l.lp === undefined) {
let lp = 0;
for (let p = 1; p <= g.prestige; p++) lp += lpGainFor(p);
g.lp = lp;
} else g.lp = Math.max(0, Math.floor(nv(l.lp, 0)));
if (window.Plant) Plant.load(g, l);                 // v8.7: THE OFFICE PLANT (+ jednorázová migrace starého Legacy shopu)
if (l.autoclicker) {
g.autoclicker.charge = nv(l.autoclicker.charge, 0);
g.autoclicker.avgCps = nv(l.autoclicker.avgCps, 0);
g.autoclicker.running = !!l.autoclicker.running;
}
sanitizeAc();
if (g.autoclicker.charge < 1 || g.autoclicker.avgCps <= 0) g.autoclicker.running = false;
if (l.stk && typeof l.stk === "object") g.stk = l.stk;                  // v9.6: akcie
if (l.pachinko && typeof l.pachinko === "object") Object.keys(l.pachinko).forEach(k => { if (/^\d+$/.test(k)) g.pachinko[k] = Math.max(0, Math.floor(nv(l.pachinko[k], 0))); });   // v9.6: i nové hodnoty míčků
g.pcPending = Array.isArray(l.pcPending) ? l.pcPending.filter(p => p && nv(p.value, 0) > 0) : [];
g.dungeon = (l.dungeon && typeof l.dungeon === "object") ? l.dungeon : {};
g.hs = (l.hs && typeof l.hs === "object") ? l.hs : {};          // HIGH STAKES (nový režim, starý dungeon se nepřenáší)
if (window.Dungeon) Dungeon.state();          // ošetří a doplní hodnoty
}

let loadTries = 0;
function loadFromServer() {
loadTries++;
google.script.run
.withSuccessHandler(loaded => {
loadTries = 0;
if (window.__irConn) window.__irConn.hide();
const bk = backupRead();
let restored = false;
if (loaded && loaded.ldap && bk && !bk.synced && bk.data && bk.nick === loaded.ldap && Number(bk.season) === Number(loaded.season) && Number(loaded.rev || 0) <= Number(bk.rev || 0)) {
try { const d = JSON.parse(bk.data); d.rev = loaded.rev || 0; d.season = loaded.season; applyLoaded(d); restored = true; } catch (e) { console.error(e); }
}
if (restored) {
log(`📴 Restored your offline progress from this device (${new Date(bk.t).toLocaleString()}) and uploading it now.`, "special");
setTimeout(function () { try { saveGame(); } catch (e) {} }, 2500);
} else if (loaded && !loaded.isNew) {
applyLoaded(loaded);
log(`Welcome back, Agent ${escapeHtml(gameState.ldap)}. Profile loaded.`, "success");
} else if (loaded && loaded.ldap) {
saveRev = Math.max(0, Math.floor(nv(loaded.rev, 0)));
gameState.ldap = loaded.ldap;
gameState.season = Number(loaded.season) || gameState.season;
log(`New terminal session for Agent ${escapeHtml(gameState.ldap)} (Season ${gameState.season}).`, "special");
}
finishLoading();
refreshLeaderboard();
})
.withFailureHandler(err => {
if (window.__irConn) {
if (loadTries < 3) { window.__irConn.show("Loading your save - attempt " + (loadTries + 1) + "/3 (the server is waking up...)"); setTimeout(loadFromServer, 1500); return; }
loadTries = 0;
window.__irConn.fail("Could not load your save: " + String(err && err.message || err), loadFromServer);
return;
}
log("Could not load your save (" + escapeHtml(String(err)) + "). Reload the page. Saving is disabled.", "error");
updateUI();
})
.loadGame();
}

function initTabGuard(start) {
try { tabChannel = new BroadcastChannel("tt_tab_guard"); } catch (e) { start(); return; }
let other = false;
tabChannel.onmessage = function (e) {
const d = e.data || {};
if (d.id === tabId) return;
if (d.t === "hello" && gameLoaded) tabChannel.postMessage({ t: "present", id: tabId });
else if (d.t === "present") other = true;
else if (d.t === "takeover" && gameLoaded) {
saveGame(function () { gameLoaded = false; showTabBlock(true); });
}
};
tabChannel.postMessage({ t: "hello", id: tabId });
setTimeout(function () { if (other) showTabBlock(true); else start(); }, 700);
}
let versionTimer = null, reloadAt = 0, reloadWatch = null;

function checkVersion() {

}).getAppVersion();
}
function scheduleReload() {

reloadAt = Date.now() + 15000 + Math.random() * 105000;      // rozptyl 15-120 s, ať se Sheets nezahltí
const banner = document.getElementById("update-banner");
if (banner) banner.classList.add("open");
reloadWatch = setInterval(function () {
const left = Math.ceil((reloadAt - Date.now()) / 1000);
const txt = left > 0 ? `New version available - refreshing in ${left}s (your game is saved first)`
: (TT.isBusy() ? "New version available - waiting for your round to finish..." : "Saving and refreshing...");
const b = document.getElementById("update-banner-text");
if (b) b.innerText = txt;
if (left <= 0 && !TT.isBusy()) {
clearInterval(reloadWatch);
let done = false;
const go = function () { if (done) return; done = true; reloading = true; location.reload(); };
saveGame(go);
setTimeout(go, 8000);
}
}, 1000);
}
