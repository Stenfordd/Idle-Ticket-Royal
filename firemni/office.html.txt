function registerAutoclickerCharge(now) {
const ac = gameState.autoclicker;

if (ac.lastClick && now - ac.lastClick > AC_IDLE_GAP) {
ac.clicks = 0;               // série přerušena -> rozpracované kliky se zahodí
ac.activeMs = 0;
} else if (ac.lastClick) {
ac.activeMs += now - ac.lastClick;
}
ac.lastClick = now;
ac.clicks += 1;
if (ac.clicks >= AC_CLICKS_PER_SEC) {
const blocks = Math.floor(ac.clicks / AC_CLICKS_PER_SEC);
const secs = ac.activeMs / 1000;
if (secs >= 1) {
const sessionAvg = Math.min(AC_MAX_CPS, ac.clicks / secs);
const earned = Math.min(blocks, AC_MAX_CHARGE - ac.charge);
if (earned > 0) {
ac.avgCps = ((ac.avgCps * ac.charge) + sessionAvg * earned) / (ac.charge + earned);
ac.charge += earned;
ringCharged(earned);
if (ac.charge >= AC_MAX_CHARGE) log("Autoclicker battery FULL (" + AC_MAX_CHARGE + "s).", "special");
}
}
const total = ac.clicks;
const rest = total - blocks * AC_CLICKS_PER_SEC;
ac.activeMs = total > 0 ? ac.activeMs * (rest / total) : 0;
ac.clicks = rest;
}
sanitizeAc();
}

/** Sezóna 2+: od levelu 20 dvojnásob, pak každých 10 levelů ×1.5 (lvl 30 = ×3, lvl 40 = ×4.5 ...) */
function chestLevelCostFactor(lvl) {
return lvl < 20 ? 1 : Math.pow(1.25, 1 + Math.floor((lvl - 20) / 10));   // v8.2: ×1.25 každých 10 levelů od LV20 (dřív 2×1.5^n)

function getChestScrapCost() {

return Math.max(10, Math.floor(base));
}
const T5_UNLOCK_LVL = 40;          // Mythic padá až od bedny lvl 40
const T6_UNLOCK_LVL = 50;          // Quantum (T6) od bedny lvl 50, jen pro části těla odemčené bossem v HIGH STAKES
function getChestOdds(lvl) {
const l = Math.max(1, lvl);
const trash = Math.max(0, 45.0 - (l - 1) * 0.5);
const t2 = Math.min(50.0, 9.8 + (l - 1) * 0.8);
const t3 = Math.min(25.0, 0.20 + (l - 1) * 0.15);
const t4 = Math.min(10.0, 0.01 + (l - 1) * 0.03);
const t5 = l < T5_UNLOCK_LVL ? 0 : Math.min(0.75, 0.025 + (l - T5_UNLOCK_LVL) * 0.003);   // v8.2: poloviční šance na Mythic
const t6n = window.Dungeon && Dungeon.t6Slots ? Dungeon.t6Slots().length : 0;
const t6 = l < T6_UNLOCK_LVL || !t6n ? 0 : Math.min(0.45, (0.005 + (l - T6_UNLOCK_LVL) * 0.002) * (window.Plant ? Plant.fx("t6") : 1)) * t6n / 6;
const t1 = Math.max(0, 100.0 - (trash + t2 + t3 + t4 + t5 + t6));
return { trash, t1, t2, t3, t4, t5, t6 };
}
/** 0 = trash, 1..6 = tier */
function rollTier(includeTrash) {
const o = getChestOdds(gameState.chestLevel);
const w = [includeTrash ? o.trash : 0, o.t1, o.t2, o.t3, o.t4, o.t5, o.t6];
const sum = w.reduce((a, b) => a + b, 0);
let r = Math.random() * sum;
for (let i = 0; i < w.length; i++) {
if (r < w[i]) return i;
r -= w[i];
}
return 6;
}

/** v8.7.3: skupiny se ⚒ itemy - 3 stejné (slot, tier, merge level) bez ohledu na ⚒, počítá se i nasazený item.
*  Výsledek: merge level +1 a nejvyšší ⚒ ze tří, o 1 stupeň menší. MERGE ALL je přeskakuje. */
function enhMergeGroups() {
const G = {};
const add = (k, u) => { (G[k] = G[k] || { units: [], slot: u.slot, tier: u.tier, lvl: u.lvl }).units.push(u); };
gameState.inventory.forEach(st => {
if ((st.lvl || 0) >= MERGE_MAX_LVL) return;
const k = st.slot + "|" + st.tier + "|" + (st.lvl || 0);
for (let i = 0; i < Math.min(st.n, 3); i++) add(k, { src: "inv", key: stackKey(st), e: enhOf(st), slot: st.slot, tier: st.tier, lvl: st.lvl || 0 });
});
SLOTS.forEach(sl => {
const it = gameState.loadout[sl]; if (!it || (it.lvl || 0) >= MERGE_MAX_LVL) return;
add(sl + "|" + it.tier + "|" + (it.lvl || 0), { src: "eq", e: enhOf(it), slot: sl, tier: it.tier, lvl: it.lvl || 0 });
});
return Object.keys(G).map(k => {
const g = G[k]; g.key = k;
if (g.units.length < MERGE_COUNT || !g.units.some(u => u.e)) return null;
const best = g.units.slice().sort((x, y) => y.e.length - x.e.length || (y.src === "eq") - (x.src === "eq"))[0];
const rest = g.units.filter(u => u !== best).sort((x, y) => x.e.length - y.e.length || (x.src === "eq") - (y.src === "eq")).slice(0, MERGE_COUNT - 1);
g.use = [best].concat(rest); g.best = best; g.eq = g.use.some(u => u.src === "eq");
g.lost = rest.reduce((a, u) => a + u.e.length, 0);
return g;
}).filter(Boolean).sort((a, b) => (b.tier - a.tier) || (b.lvl - a.lvl));
}
function mergeEnh(gkey) {
const g = enhMergeGroups().find(x => x.key === gkey); if (!g) return;
const E = g.best.e.length, name = itemName(g);
if (!confirm(`Merge 3× ${name} +${g.lvl}${g.eq ? " (incl. your equipped one)" : ""}?\nResult: +${g.lvl + 1} with ⚒${E} → ⚒${Math.max(0, E - 1)}${g.lost ? "\nThe other ⚒ stages (" + g.lost + ") are used up." : ""}`)) return;
g.use.forEach(u => { if (u.src === "inv") invTake(u.key, 1); });
if (g.eq) gameState.loadout[g.slot] = null;
const roll = Math.random() * 100;
let res;
if (roll < 5) {                                       // fail: nejlepší item přežije beze změny, ostatní dva se spotřebují
res = { slot: g.slot, tier: g.tier, lvl: g.lvl }; if (g.best.e) res.e = g.best.e;
const refund = g.use.slice(1).reduce((a, u) => a + Math.floor(itemScrapValue({ tier: g.tier, lvl: g.lvl }) * 0.5), 0);
gameState.scrap += refund;
log(`MERGE FAILED: ${name} +${g.lvl} ⚒${E} survived, the other two were lost (+${refund} scrap salvaged).`, "error");
} else {
const bonus = roll < 15 ? 2 : 1;
res = { slot: g.slot, tier: g.tier, lvl: Math.min(MERGE_MAX_LVL, g.lvl + bonus) };
const e = g.best.e.slice(0, -1); if (e) res.e = e;
log(`⚒ MERGE: ${name} → +${res.lvl}${bonus === 2 ? " (CRIT +2)" : ""} with ⚒${e.length}.`, "special");
}
if (g.eq) gameState.loadout[g.slot] = res;
else invAdd(res.slot, res.tier, res.lvl, 1, res.e);
updateUI(); TT.saveSoon();
}
