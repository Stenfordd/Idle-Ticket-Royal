function mergeAll() {
let ok = 0, crit = 0, fail = 0, refund = 0, rounds = 0;
for (; rounds < 60; rounds++) {
const s = gameState.inventory.filter(x => x.n >= MERGE_COUNT && (x.lvl || 0) < MERGE_MAX_LVL && !enhOf(x))
.sort((a, b) => (a.lvl || 0) - (b.lvl || 0))[0];
if (!s) break;
const r = mergeStack(s, Math.floor(s.n / MERGE_COUNT));
ok += r.ok; crit += r.crit; fail += r.fail; refund += r.refund;
}
if (!ok && !fail) { log("Nothing to merge.", "warning"); return; }
gameState.scrap += refund;
log(`MERGE ALL: ${ok} merges${crit ? " (" + crit + " crit +2)" : ""}${fail ? ", " + fail + " failed (+" + refund + " scrap)" : ""}.`, "special");
updateUI();
TT.saveSoon();
}
function mergeStack(s0, times) {
const key = stackKey(s0);
const slot = s0.slot, tier = s0.tier, lvl = s0.lvl || 0;
const r = { ok: 0, crit: 0, fail: 0, refund: 0, done: 0 };
for (let i = 0; i < times; i++) {
if (!invTake(key, MERGE_COUNT)) break;
r.done++;
const roll = Math.random() * 100;
if (roll < 5) {
r.fail++;
r.refund += Math.floor(itemScrapValue({ tier: tier, lvl: lvl }) * MERGE_COUNT * 0.5);
} else {
const bonus = roll < 15 ? 2 : 1;
if (bonus === 2) r.crit++;
invAdd(slot, tier, Math.min(MERGE_MAX_LVL, lvl + bonus), 1);
r.ok++;
}
}
return r;
}
function performSmartMerge(key, times) {
const s0 = invByKey(key);

done++;
const roll = Math.random() * 100;
if (roll < 5) {
fail++;
refund += Math.floor(itemScrapValue({ tier: tier, lvl: lvl }) * MERGE_COUNT * 0.5);

const bonus = roll < 15 ? 2 : 1;
if (bonus === 2) crit++;
invAdd(slot, tier, Math.min(MERGE_MAX_LVL, lvl + bonus), 1);
ok++;
}
}
if (done === 0) { log(`Cannot merge: need ${MERGE_COUNT} identical items.`, "warning"); return; }
gameState.scrap += refund;
const name = itemName({ slot: slot, tier: tier });
log(`MERGE ${name} +${lvl}: ${ok} success${crit ? " (" + crit + " crit +2)" : ""}${fail ? ", " + fail + " failed (+" + refund + " scrap salvaged)" : ""}.`, crit || ok ? "special" : "error");

}
