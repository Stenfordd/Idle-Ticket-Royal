function pcLaunch(value, batch, delay) {

const bin = (o + PC_ROWS) / 2;
const pts = [{ x: W / 2 + (R() - 0.5) * 4, y: 6, peg: null }];
for (let i = 0; i < PC_ROWS; i++) pts.push({ x: pegX(i, offs[i]) + (offs[i + 1] - offs[i]) * (PEG_R + BALL_R - 2) * 0.55 + (R() - 0.5) * 1.6, y: pegY(i) - PEG_R - BALL_R + 1, peg: i + ":" + offs[i] });
pts.push({ x: W / 2 + (2 * bin - PC_ROWS) * HS, y: BIN_Y + BIN_H - BALL_R - 4, peg: null, bin: bin });
const id = Date.now() + "_" + (pc.seq++);
const risk = pc.risk;
TT.state().pcPending.push({ id: id, value: value, bin: bin, risk: risk });     // míček v letu je v savu
pc.session.n++; pc.session.spent += value;
const t0 = performance.now() + (delay || 0);
const segs = pts.length - 1;

pcKick();
}

/** pozice míčku: vodorovně plynule, svisle parabola (vyskočí a gravitací spadne na další kolík) */
function pcBallPos(b, now) {
let el = now - b.t0;
if (el < 0) return null;
let seg = 0;
while (seg < b.durs.length - 1 && el > b.durs[seg]) { el -= b.durs[seg]; seg++; }
const f = Math.min(1, el / b.durs[seg]);
const p = b.pts[seg], q = b.pts[seg + 1];
const ex = f < 0.5 ? 2 * f * f : 1 - Math.pow(-2 * f + 2, 2) / 2;
const x = p.x + (q.x - p.x) * (seg === 0 ? f : ex);
const hop = seg === 0 ? 0 : (seg === b.durs.length - 1 ? 10 : 9);

return { x: x, y: y, seg: seg };
}

function pcTick() {
const now = performance.now();

}

function roundRect(ctx, x, y, w, h, r) {
