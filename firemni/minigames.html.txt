const cr = { phase: "idle", jumps: 0, bet: 0, deadAt: -1 };
function crRender() {
const road = $("cr-road");
if (!road) return;
let html = "";
for (let i = 0; i <= CR_MAX_JUMPS; i++) {
let cls = "cr-cell", content = "";
if (i === 0) cls += " curb";
if (i > 0 && i <= cr.jumps) cls += " done";
if (cr.phase === "run" && i === cr.jumps + 1) cls += " next";
if (cr.phase === "dead") {
if (i === cr.deadAt) { cls += " dead"; content = "💥"; }
else if (i > 0 && i <= cr.jumps) content = "✔";
else if (i > cr.deadAt) content = `<span class="cr-car" style="left:${6 + (i * 7) % 18}px">${i % 2 ? "🚗" : "🚚"}</span>`;
} else {
if (i === cr.jumps) content = `<span class="cr-chicken" id="cr-chicken">🐔</span>`;
else if (i > 0 && i < cr.jumps) content = "✔";
else if (i > cr.jumps) content = `<span class="cr-car" style="left:${6 + (i * 7) % 18}px">${i % 2 ? "🚗" : "🚚"}</span>`;
}
html += `<div class="${cls}">${content}<span class="cr-lbl">${i === 0 ? "START" : crMult(i) + "×"}</span></div>`;
}
road.innerHTML = html;
const idx = cr.phase === "dead" ? cr.deadAt : cr.jumps;
const here = road.children[idx];
if (here) road.scrollLeft = Math.max(0, here.offsetLeft - road.clientWidth / 2 + 23);
const m = cr.phase === "dead" ? 0 : crMult(cr.jumps);
$("cr-jumps").innerText = cr.phase === "dead" ? cr.jumps : cr.jumps;
$("cr-mult").innerText = m.toFixed(2) + "×";
$("cr-payout").innerText = cr.phase === "run" ? fmt(cr.bet * m) : "0";
$("cr-next-mult").innerText = crMult(Math.min(CR_MAX_JUMPS, cr.jumps + 1)).toFixed(2);
$("cr-start").disabled = cr.phase === "run";
$("cr-jump").disabled = cr.phase !== "run";
$("cr-cash").disabled = cr.phase !== "run" || cr.jumps < 1;
}
function crStart() {
if (cr.phase === "run") return;
const input = $("cr-bet");
const bet = TT.cleanBet(input.value, 10, TT.maxBet ? TT.maxBet() : CR_MAX_BET);
input.value = bet;
if (!TT.spend(bet)) { $("cr-msg").style.color = "var(--crimson)"; $("cr-msg").innerText = "Not enough GC."; return; }
cr.phase = "run"; cr.jumps = 0; cr.bet = bet; cr.deadAt = -1;
$("cr-msg").style.color = "var(--cyan)";
$("cr-msg").innerText = "The road is open. Jump!";
crRender();
}
function crJump() {
if (cr.phase !== "run") return;
if (TT.rnd() < 0.5) {                                   // 50 % smrt
cr.phase = "dead"; cr.deadAt = cr.jumps + 1;
$("cr-msg").style.color = "var(--crimson)";
$("cr-msg").innerText = `SPLAT! Hit on jump ${cr.deadAt}. Lost ${fmt(cr.bet)} GC.`;
TT.log(`CHICKEN ROAD: hit by a car on jump ${cr.deadAt}. Lost ${fmt(cr.bet)} GC.`, "error");
crRender();
return;
}
cr.jumps += 1;
crRender();
const ch = $("cr-chicken");
if (ch) { ch.classList.remove("hop"); void ch.offsetWidth; ch.classList.add("hop"); }
$("cr-msg").style.color = "var(--emerald)";
$("cr-msg").innerText = `Safe! ${crMult(cr.jumps).toFixed(2)}× (${fmt(cr.bet * crMult(cr.jumps))} GC)`;
if (cr.jumps >= CR_MAX_JUMPS) crCash(true);
}
function crCash(auto) {
if (cr.phase !== "run" || cr.jumps < 1) return;
const mult = crMult(cr.jumps);
const payout = Math.floor(cr.bet * mult);
const jumps = cr.jumps, bet = cr.bet;
cr.phase = "idle"; cr.jumps = 0;
TT.win(payout);
TT.log(`CHICKEN ROAD CASH OUT: ${jumps} jumps, ${mult.toFixed(2)}× → +${fmt(payout)} GC${auto ? " (max jumps!)" : ""}`, "special");
if (mult >= CR_JACKPOT_MULT) TT.announce("CHICKEN x" + mult.toFixed(0), "ROAD", bet, payout);
crRender();
$("cr-msg").style.color = "var(--amber)";
$("cr-msg").innerText = `CASHED OUT ${mult.toFixed(2)}× = +${fmt(payout)} GC!`;
}

function pcDrop(n) {

const batch = n > 1 ? "b" + (pc.seq++) : null;
for (let i = 0; i < n; i++) {
if (stock[key] <= 0 || pc.balls.length >= PC_MAX_ACTIVE) break;
stock[key] -= 1;
pcLaunch(Number(key), batch, i * 110);
dropped++;
}
if (batch && dropped > 1) pc.batches[batch] = { n: dropped, left: dropped, spent: dropped * Number(key), back: 0, best: 0 };
if (dropped === 0) {
$("pc-msg").style.color = "var(--amber)";

if (pc.auto && stock[key] <= 0) pcAuto();
}
pcRefresh();
TT.saveSoon();
return dropped;
}

function pegX(row, k) { return W / 2 + k * HS; }      // k = posun (-row..row po 2)
function pegY(row) { return TOP + row * DY; }
function pcLaunch(value, batch, delay) {
const R = () => TT.rnd();
const offs = [0];
let o = 0;
for (let i = 0; i < PC_ROWS; i++) { o += (R() < 0.5) ? -1 : 1; offs.push(o); }
const bin = (o + PC_ROWS) / 2;
const pts = [{ x: W / 2 + (R() - 0.5) * 4, y: 6, peg: null }];
for (let i = 0; i < PC_ROWS; i++) pts.push({ x: pegX(i, offs[i]) + (offs[i + 1] - offs[i]) * (PEG_R + BALL_R - 2) * 0.55 + (R() - 0.5) * 1.6, y: pegY(i) - PEG_R - BALL_R + 1, peg: i + ":" + offs[i] });
pts.push({ x: W / 2 + (2 * bin - PC_ROWS) * HS, y: BIN_Y + BIN_H - BALL_R - 4, peg: null, bin: bin });
const id = Date.now() + "_" + (pc.seq++);
const risk = pc.risk;
TT.state().pcPending.push({ id: id, value: value, bin: bin, risk: risk });     // míček v letu je v savu
pc.session.n++; pc.session.spent += value;
const t0 = performance.now() + (delay || 0);
const segs = pts.length - 1;
const durs = [];
for (let i = 0; i < segs; i++) durs.push(i === 0 ? 260 : (i === segs - 1 ? 300 : SEG_MS * (0.9 + R() * 0.2)));
pc.balls.push({ id: id, value: value, bin: bin, risk: risk, pts: pts, durs: durs, t0: t0, dur: durs.reduce((a, b) => a + b, 0), batch: batch, trail: [], lastSeg: -1, hue: 40 + R() * 20 });
pcKick();
}
function pcKick() { if (!pc.raf) pc.raf = requestAnimationFrame(pcFrame); }
/** pozice míčku: vodorovně plynule, svisle parabola (vyskočí a gravitací spadne na další kolík) */
function pcBallPos(b, now) {
let el = now - b.t0;
if (el < 0) return null;
let seg = 0;
while (seg < b.durs.length - 1 && el > b.durs[seg]) { el -= b.durs[seg]; seg++; }
const f = Math.min(1, el / b.durs[seg]);
const p = b.pts[seg], q = b.pts[seg + 1];
const ex = f < 0.5 ? 2 * f * f : 1 - Math.pow(-2 * f + 2, 2) / 2;
const x = p.x + (q.x - p.x) * (seg === 0 ? f : ex);
const hop = seg === 0 ? 0 : (seg === b.durs.length - 1 ? 10 : 9);
const y = p.y + (q.y - p.y) * f * f - hop * 4 * f * (1 - f) * (1 - f * 0.3);
return { x: x, y: y, seg: seg };
}
/** Logika přistání - nezávislá na vykreslování (běží i v neviditelném tabu přes watchdog). */
function pcTick() {
const now = performance.now();
const rest = [];
for (const b of pc.balls) {
if (now - b.t0 >= b.dur) { try { pcLand(b); } catch (e) { console.error(e); } }
else rest.push(b);
}
pc.balls = rest;
pc.flash = pc.flash.filter(f => now - f.t < 700);
if (!pc.balls.length && !pc.flash.length) pc.raf = 0;
pcRefresh();
}
function pcFrame(now) {
try {
pcTick();
const ctx = pc.ctx;
pcDrawBoard(ctx, now);
for (const b of pc.balls) {
const pos = pcBallPos(b, now);
if (!pos) continue;
if (pos.seg !== b.lastSeg) {               // nový segment = náraz do kolíku
b.lastSeg = pos.seg;
const pg = b.pts[pos.seg].peg;
if (pg) pc.pegHits[pg] = now;
}
b.trail.push([pos.x, pos.y]); if (b.trail.length > 7) b.trail.shift();
for (let i = 0; i < b.trail.length - 1; i++) {
const t = b.trail[i], a = (i + 1) / b.trail.length;
ctx.beginPath(); ctx.arc(t[0], t[1], BALL_R * a * 0.8, 0, Math.PI * 2);
ctx.fillStyle = `hsla(${b.hue},100%,60%,${0.25 * a})`; ctx.fill();
}
const g = ctx.createRadialGradient(pos.x - 1.6, pos.y - 1.8, 0.5, pos.x, pos.y, BALL_R);
g.addColorStop(0, "#fffbe6"); g.addColorStop(0.45, `hsl(${b.hue},100%,58%)`); g.addColorStop(1, `hsl(${b.hue - 15},90%,32%)`);
ctx.beginPath(); ctx.arc(pos.x, pos.y, BALL_R, 0, Math.PI * 2);
ctx.shadowColor = `hsla(${b.hue},100%,60%,.8)`; ctx.shadowBlur = 10;
ctx.fillStyle = g; ctx.fill(); ctx.shadowBlur = 0;
}
} catch (e) { console.error(e); }
if (pc.balls.length || pc.flash.length) pc.raf = requestAnimationFrame(pcFrame);
else { pc.raf = 0; try { pcDrawBoard(pc.ctx, performance.now()); } catch (e) {} }
}

function pcLand(b) {
const st = TT.state();
st.pcPending = (st.pcPending || []).filter(p => p.id !== b.id);
const table = PC_RISKS[b.risk] || PC_RISKS.high;
const mult = table[b.bin];
const payout = Math.floor(b.value * mult);
pc.flash.push({ bin: b.bin, t: performance.now(), mult: mult });
pc.binHits[b.bin] = performance.now();
pc.session.back += payout;
if (payout > 0) TT.win(payout);
pcPop(b.bin, mult + "×", pcColor(mult), mult >= 25);
const msg = $("pc-msg");
if (mult >= 1) {
msg.style.color = mult >= 25 ? "var(--amber)" : "var(--emerald)";
msg.innerText = `Ball @${b.value} → ${mult}× = +${fmt(payout)} GC!`;
if (mult >= 10) TT.log(`PACHINKO: ball (${b.value} GC) hit ${mult}× → +${fmt(payout)} GC`, mult >= 25 ? "special" : "success");
} else {
msg.style.color = "var(--text-dim)";
msg.innerText = `Ball @${b.value} → ${mult}× (${fmt(payout)} GC back).`;
}
if (mult >= 25) TT.announce("PACHINKO x" + mult, "BALL", b.value, payout);
const bt = b.batch && pc.batches[b.batch];
if (bt) {
bt.left--; bt.back += payout; bt.best = Math.max(bt.best, mult);
if (bt.left <= 0) {
delete pc.batches[b.batch];
const net = bt.back - bt.spent;
msg.style.color = net >= 0 ? "var(--emerald)" : "var(--crimson)";
msg.innerText = `${bt.n} BALLS: spent ${fmt(bt.spent)} → got ${fmt(bt.back)} GC (${net >= 0 ? "+" : ""}${fmt(net)}) · best ${bt.best}×`;
TT.log(`PACHINKO ×${bt.n}: spent ${fmt(bt.spent)}, got back ${fmt(bt.back)} GC → net ${net >= 0 ? "+" : ""}${fmt(net)}.`, net >= 0 ? "success" : "error");
}
}
}
function pcResolvePending() {
const st = TT.state();
const list = st.pcPending || [];
if (!list.length) return;
let total = 0;
list.forEach(p => { total += Math.floor(p.value * ((PC_RISKS[p.risk] || PC_RISKS.high)[p.bin] || 0)); });
st.pcPending = [];
if (total > 0) TT.win(total);
TT.log(`Pachinko: ${list.length} ball(s) from your last session were resolved (+${fmt(total)} GC).`, "special");
}
function pcColor(m) { return m >= 100 ? "#ff4dff" : m >= 20 ? "#f43f5e" : m >= 4 ? "#fbbf24" : m >= 1.5 ? "#22c55e" : m >= 1 ? "#2563eb" : "#3b4560"; }
function setupCanvas() {

cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr);
const ctx = cv.getContext("2d");
ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
return ctx;
}
function pcDrawBoard(ctx, now) {
if (!ctx) return;
now = now || performance.now();
ctx.clearRect(0, 0, W, H);
ctx.strokeStyle = "rgba(0,240,255,.25)"; ctx.lineWidth = 2;
ctx.beginPath(); ctx.moveTo(W / 2 - 22, 0); ctx.lineTo(W / 2 - 8, 18); ctx.moveTo(W / 2 + 22, 0); ctx.lineTo(W / 2 + 8, 18); ctx.stroke();
for (let r = 0; r < PC_ROWS; r++) {
for (let k = -r; k <= r; k += 2) {
const x = pegX(r, k), y = pegY(r);
const hit = pc.pegHits[r + ":" + k];
const a = hit ? Math.max(0, 1 - (now - hit) / 350) : 0;
if (a > 0) {
ctx.beginPath(); ctx.arc(x, y, PEG_R + 6 * a, 0, Math.PI * 2);
ctx.fillStyle = `rgba(255,220,120,${0.35 * a})`; ctx.fill();

function physStep(r, P) {
r.vy -= G * DT;
const oy = r.y;
r.x += r.vx * DT; r.y += r.vy * DT;
if (r.x < 8) { r.x = 8; r.vx = -r.vx * 0.6; } else if (r.x > W - 8) { r.x = W - 8; r.vx = -r.vx * 0.6; }
for (let k = 0; k < P.length; k++) {
const p = P[k];
if (r.x + 7 < p.x || r.x - 7 > p.x + p.w) continue;
if (r.vy <= 0 && oy >= p.y && r.y <= p.y) { const fall = -r.vy; r.y = p.y; r.vy = 0; r.vx = 0; r.ground = true; return { k: k, fall: fall }; }   // přistání
if (r.vy > 0 && oy + 22 <= p.y - p.h && r.y + 22 > p.y - p.h) { r.y = p.y - p.h - 22; r.vy = 0; r.bonk = 1; }      // hlavou o stůl
}
if (r.y < 24) { const fall = -r.vy; r.y = 24; r.vy = 0; r.vx = 0; r.ground = true; return { k: 0, fall: fall }; }
return null;
}
