function prsJoin(id) {
var me = getUserLdap(), lock = prsLock_();
try {
var ts = prsLoadAll_(), t = ts.filter(function (x) { return x.id === id; })[0];
if (!t || t.status !== 'wait') return { ok: false, reason: 'gone' };
if (ts.some(function (x) { return x.players.indexOf(me) >= 0 && (x.status === 'wait' || x.status === 'play') && !(x.seen && x.seen[me]); })) return { ok: false, reason: 'already' };
if (t.players.length >= t.seats) return { ok: false, reason: 'seats' };
t.players.push(me); t.rev++; prsLog_(t, me + ' sat down');
if (t.players.length >= t.seats) prsDeal_(t);
prsPut_(t); prsSetIndex_(prsIndex_());
return { ok: true, id: t.id, stake: t.stake };
} finally { lock.releaseLock(); }
}
function prsStart(id) {
var me = getUserLdap(), lock = prsLock_();
try {
var t = prsGet_(id);
if (!t || t.status !== 'wait') return { ok: false, reason: 'gone' };
if (t.host !== me) return { ok: false, reason: 'host' };
if (t.players.length < 2) return { ok: false, reason: 'alone' };
prsDeal_(t); prsPut_(t); prsSetIndex_(prsIndex_());
return { ok: true };
} finally { lock.releaseLock(); }
}
function prsLeave(id) {
var me = getUserLdap(), lock = prsLock_();
try {
var t = prsGet_(id);
if (!t || t.players.indexOf(me) < 0) return { ok: true };
if (t.status === 'wait') {
t.players = t.players.filter(function (n) { return n !== me; });
prsPay_(me, t.stake, 'REFUND', t.id); prsLog_(t, me + ' left (stake refunded)');
if (!t.players.length) { prsDel_(id); prsSetIndex_(prsIndex_().filter(function (x) { return x !== id; })); return { ok: true }; }
if (t.host === me) t.host = t.players[0];
} else if (t.status === 'play' && (t.fin || []).indexOf(me) >= 0) { t.seen = t.seen || {}; t.seen[me] = 1; }
else if (t.status === 'play') prsRemovePlayer_(t, me, 'left the table');
else if (t.status === 'done') { t.seen = t.seen || {}; t.seen[me] = 1; }
t.rev++; prsPut_(t); prsSetIndex_(prsIndex_());
return { ok: true };
} finally { lock.releaseLock(); }
}

var pay = [];

t = prsGet_(id);

}
/** tah: { t: 'play', c: karta, suit: barva pro svrška } | { t: 'draw' } (líznout / vzít trest / stát na eso) */
function prsMove(id, mv) {
var me = getUserLdap(), lock = prsLock_();
try {
var t = prsGet_(id);
if (!t || t.status !== 'play') return { ok: false, reason: 'gone' };
prsTimeouts_(t);
if (t.status !== 'play') { prsPut_(t); return { ok: false, reason: 'gone', view: prsView_(t, me) }; }
if (t.order[t.turn] !== me) { prsPut_(t); return { ok: false, reason: 'turn', view: prsView_(t, me) }; }
mv = mv || {};
var hand = t.hands[me];
if (mv.t === 'play') {
var c = Number(mv.c), i = hand.indexOf(c);
if (i < 0 || !prsPlayable_(t, c)) return { ok: false, reason: 'card', view: prsView_(t, me) };
hand.splice(i, 1); t.pile.push(c);
var r = prsRank_(c);
if (r === 5) { var su = Number(mv.suit); t.suit = su >= 0 && su <= 3 ? su : prsSuit_(c); }
else t.suit = prsSuit_(c);
if (r === 0) t.pen = (t.pen || 0) + 2;
if (r === 7) t.skip = true;
t.miss[me] = 0; t.moved = Date.now();
prsLog_(t, me + ' plays ' + prsName_(c) + (r === 5 ? ' and picks ' + PRS_SUITS[t.suit] : r === 0 ? ' (+' + t.pen + ' for the next player)' : r === 7 ? ' (next player stands)' : ''));
if (!hand.length) { prsPlaceOut_(t, me, 'out of cards'); t.rev++; prsPut_(t); return { ok: true, view: prsView_(t, me) }; }
prsNext_(t);
} else {
t.miss[me] = 0;
prsAutoDraw_(t, me, false);
}
t.rev++; prsPut_(t);
return { ok: true, view: prsView_(t, me) };
} finally { lock.releaseLock(); }
}
var CK_MAX_TABLES = 3, CK_TURN_MS = 45000, CK_WAIT_MS = 180000, CK_IDLE_MS = 30 * 60000, CK_RAKE = 0.05, CK_MAX_MISS = 3, CK_QUIET = 40;

function ckNewBoard_() { var b = []; for (var i = 0; i < 64; i++) { var r = Math.floor(i / 8), c = i % 8; b.push((r + c) % 2 === 1 ? (r < 3 ? -1 : r > 4 ? 1 : 0) : 0); } return b; }
/** všechny legální tahy hráče p (1 bílý, -1 černý) jako cesty [od, pole1, pole2, ...] + seznam sebraných */
function ckMoves_(b, p) {
var caps = [], kingCaps = [], simple = [];
for (var sq = 0; sq < 64; sq++) {
var pc = b[sq]; if (!pc || (pc > 0) !== (p > 0)) continue;
var king = Math.abs(pc) === 2, found = [];
ckCaps_(b, sq, sq, pc, king, [sq], [], found);
if (found.length) { if (king) kingCaps = kingCaps.concat(found); else caps = caps.concat(found); continue; }
var r = Math.floor(sq / 8), c = sq % 8, dirs = king ? [[-1, -1], [-1, 1], [1, -1], [1, 1]] : [[p > 0 ? -1 : 1, -1], [p > 0 ? -1 : 1, 1]];
dirs.forEach(function (d) {
for (var k = 1; k < 8; k++) {
var rr = r + d[0] * k, cc = c + d[1] * k; if (rr < 0 || rr > 7 || cc < 0 || cc > 7 || b[rr * 8 + cc]) break;
simple.push({ path: [sq, rr * 8 + cc], cap: [] }); if (!king) break;
}
});
}
if (kingCaps.length) return kingCaps;
if (caps.length) return caps;
return simple;
}
function ckCaps_(b, origin, sq, pc, king, path, cap, out) {
var p = pc > 0 ? 1 : -1, r = Math.floor(sq / 8), c = sq % 8, any = false;
var dirs = king ? [[-1, -1], [-1, 1], [1, -1], [1, 1]] : [[p > 0 ? -1 : 1, -1], [p > 0 ? -1 : 1, 1]];
var occ = function (i) { return i === origin ? 0 : b[i]; };
dirs.forEach(function (d) {
var k = 1, rr, cc, i;
if (king) { while (true) { rr = r + d[0] * k; cc = c + d[1] * k; if (rr < 0 || rr > 7 || cc < 0 || cc > 7) return; i = rr * 8 + cc; if (occ(i)) break; k++; } }
else { rr = r + d[0]; cc = c + d[1]; if (rr < 0 || rr > 7 || cc < 0 || cc > 7) return; i = rr * 8 + cc; }
var v = occ(i);
if (!v || (v > 0) === (p > 0) || cap.indexOf(i) >= 0) return;            // musí to být soupeř, ještě nesebraný
for (var m = 1; m < 8; m++) {
var lr = rr + d[0] * m, lc = cc + d[1] * m; if (lr < 0 || lr > 7 || lc < 0 || lc > 7) break;
var li = lr * 8 + lc; if (occ(li)) break;
any = true;
var promo = !king && (p > 0 ? lr === 0 : lr === 7);
var np = path.concat([li]), nc = cap.concat([i]);
if (promo) out.push({ path: np, cap: nc });                                // povýšení ukončí tah
else { var before = out.length; ckCaps_(b, origin, li, pc, king, np, nc, out); if (out.length === before) out.push({ path: np, cap: nc }); }
if (!king) break;
}
});
return any;
}
function ckApply_(b, mv) {
var from = mv.path[0], to = mv.path[mv.path.length - 1], pc = b[from];
b[from] = 0; mv.cap.forEach(function (i) { b[i] = 0; });
var r = Math.floor(to / 8);
if (Math.abs(pc) === 1 && ((pc > 0 && r === 0) || (pc < 0 && r === 7))) pc = pc * 2;
b[to] = pc;
}

function ckEnd_(t, winner, why) {
t.status = 'done'; t.doneAt = Date.now(); t.deadline = 0; t.winner = winner || ''; t.why = why;
if (winner) {
var win = Math.floor(t.stake * 2 * (1 - CK_RAKE)); t.win = win; ckPay_(winner, win, 'WIN', t.id);
ckLog_(t, '🏆 ' + winner + ' wins ' + win.toLocaleString() + ' GC (' + why + ')');
try { getDb_().chat.appendRow([new Date(), '⛀ CHECKERS', '🏆 ' + winner + ' beat ' + (winner === t.white ? t.black : t.white) + ' at checkers (+' + win.toLocaleString() + ' GC, ' + why + ').']); } catch (e) {}
} else {
[t.white, t.black].forEach(function (n) { ckPay_(n, t.stake, 'DRAW', t.id); });
ckLog_(t, '🤝 Draw (' + why + ') - stakes returned');
}
}
function ckAfterMove_(t) {
t.turnColor = -t.turnColor; t.deadline = Date.now() + CK_TURN_MS; t.moved = Date.now(); t.rev++;
var next = t.turnColor > 0 ? t.white : t.black, moves = ckMoves_(t.board, t.turnColor);
if (!moves.length) { ckEnd_(t, next === t.white ? t.black : t.white, next + ' has no moves left'); return; }
if (t.quiet >= CK_QUIET) ckEnd_(t, '', CK_QUIET + ' moves without a capture');
}
function ckDoMove_(t, mv, nick, auto) {
ckApply_(t.board, mv);
t.last = { path: mv.path, cap: mv.cap, by: nick };
t.quiet = mv.cap.length ? 0 : (t.quiet || 0) + 1;
t.moves = (t.moves || 0) + 1; t.drawOffer = '';
ckLog_(t, (auto ? '⏱ ' : '') + nick + (mv.cap.length ? ' captures ' + mv.cap.length + '×' : ' moves') + (auto ? ' (time ran out, random move)' : ''));
ckAfterMove_(t);
}
function ckTimeouts_(t) {
var ch = false, g = 0;
while (t.status === 'play' && t.deadline && Date.now() > t.deadline && g++ < 4) {
var nick = t.turnColor > 0 ? t.white : t.black;
t.miss[nick] = (t.miss[nick] || 0) + 1; ch = true;
if (t.miss[nick] >= CK_MAX_MISS) { ckEnd_(t, nick === t.white ? t.black : t.white, nick + ' timed out 3×'); t.rev++; break; }
var ms = ckMoves_(t.board, t.turnColor); ckDoMove_(t, ms[Math.floor(Math.random() * ms.length)], nick, true);
}
return ch;
}
function ckHousekeep_(t) {
var now = Date.now();
if (t.status === 'wait' && now - t.created > CK_WAIT_MS) { ckPay_(t.host, t.stake, 'REFUND', t.id); t.status = 'gone'; return true; }
if (t.status === 'done' && now - t.doneAt > 120000) { t.status = 'gone'; return true; }
if (t.status === 'play' && now - t.moved > CK_IDLE_MS) { ckEnd_(t, '', 'abandoned'); t.status = 'gone'; return true; }
return false;
}
function ckLoadAll_() {

if (hk || to) { dirty = true; if (t.status === 'gone') { ckDel_(id); return; } ckPut_(t); }
keep.push(id); out.push(t);
});
if (dirty) ckSetIndex_(keep);
return out;
}

/** tah = cesta polí [od, ..., kam]; musí přesně odpovídat jednomu z legálních tahů */

}
/** remíza: what = 'offer' | 'accept' | 'decline' */
function ckDraw(id, what) {
var me = getUserLdap(), lock = prsLock_();
try {
var t = ckGet_(id); if (!t || t.status !== 'play' || !ckColor_(t, me)) return { ok: false, reason: 'gone' };

else if (what === 'decline' && t.drawOffer && t.drawOffer !== me) { t.drawOffer = ''; ckLog_(t, me + ' declines the draw'); }
t.rev++; ckPut_(t);
return { ok: true, view: ckView_(t, me) };
} finally { lock.releaseLock(); }
}

* Tahový souboj: hráč udeří (útok se losuje z tabulky podle třídy a postoje), pak boss.

function rng(seed) {
let a = (seed >>> 0) || 1;
return function () {
a |= 0; a = (a + 0x6D2B79F5) | 0;
let t = Math.imul(a ^ (a >>> 15), 1 | a);
t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};

if (kind === "bf") {
self = p.atk * BACKFIRE_SELF[p.cls] * (1 + p.fx.bfDmg);
if (p.cls === "engineer" && p.tier >= 2) self *= 0.7;
if (p.cls === "sysadmin" && p.tier >= 3) { self *= 0.5; f.shield += Math.round(self); }
}
if (kind === "jp" && p.cls === "sysadmin" && p.tier >= 4 && p.branch === "b") dmg += f.shield * 2;
const dealt = damageBoss(f, dmg);
ev(f, "pattack", dealt, { kind: kind, mult: Math.round(mult * 100) / 100, stance: stance });
if (self > 0) damagePlayer(f, self, "backfire");
if (kind === "jp" && p.cls === "cyber" && p.tier >= 3) { f.dot = 3; ev(f, "mech", 0, { msg: "ZERO-DAY PAYLOAD injected!" }); }
if (p.cls === "sysadmin" && p.tier >= 2 && stance === "safe" && mult >= 1) { const h = Math.round(p.maxHp * 0.04); f.php = Math.min(p.maxHp, f.php + h); ev(f, "heal", h); }
if (p.cls === "engineer" && p.tier >= 3) f.ciStacks = mult >= 1 ? Math.min(4, f.ciStacks + 1) : 0;
f.prevMult = mult >= 1 ? mult : 0;
if (mech === "excel" && dealt > 0 && R() < 0.15) damagePlayer(f, dealt * 0.3, "reflect");
if (mech === "broker" && mult < 0.8) f.weakStacks = Math.min(5, f.weakStacks + 1);
if (mech === "s3" && mult >= 1.5) { f.bshield += Math.round(dealt * 0.2); ev(f, "mech", 0, { msg: "DATA SIPHON - boss gains a shield!" }); }
if (mech === "ransom" && mult >= 1.3 && f.ransom) { f.ransom = false; ev(f, "mech", 0, { msg: "Decrypted! ATK restored." }); }
if (mech === "hr" && (kind === "jp" || mult >= 1.6)) { f.forceSafe = 2; ev(f, "mech", 0, { msg: "MANDATORY TEAM BUILDING - forced into SAFE!" }); }
if (mech === "llm" && (kind === "jp" || mult >= 1.6) && f.bhp > 0) { ev(f, "mech", 0, { msg: "HALLUCINATION COUNTER!" }); bossHit(f, b.atk * (0.5 + R()), "counter", stance); }
}
if (f.php <= 0) return finish(f, false, start);
if (f.bhp <= 0) return finish(f, true, start);
let skip = false;
if (mech === "router" && f.turn % 4 === 0) { skip = true; ev(f, "mech", 0, { msg: "Signal dropped - the router skips its attack." }); }
if (!skip) {
if (mech === "ddos") { for (let i = 0; i < 3; i++) bossHit(f, b.atk * 0.4, "packet", stance); }
else {
let a = b.atk;
if (mech === "dust") a *= 0.5 + 0.5 * (f.bhp / f.bmax);
if (mech === "cron") a *= (f.turn % 4 === 0 ? 2.6 : 0.6);
if (mech === "leak") a *= 1 + 0.08 * (f.turn - 1);
if (mech === "api" && f.bhp < f.bmax * 0.5) a *= 1.4;
if (mech === "darkweb" && f.bhp < f.bmax * 0.3) a *= 2;
if (mech === "mesh" && f.hitStreak >= 3) { a *= 2; f.hitStreak = 0; ev(f, "mech", 0, { msg: "CASCADE FAILURE!" }); }
if (mech === "cto" && f.ctoWant && stance !== f.ctoWant) a *= 1.6;
bossHit(f, a, "hit", stance);
}
}
if (mech === "cobol" && f.turn % 4 === 0) { const h = Math.round(f.bmax * 0.12); f.bhp = Math.min(f.bmax, f.bhp + h); ev(f, "bheal", h); }
if (mech === "darkweb" && f.bhp < f.bmax * 0.3) { const d = Math.round(f.bmax * 0.05); f.bhp -= d; ev(f, "mech", d, { msg: "Self-destruct: the boss burns " + d + " HP." }); }
if (f.lockLeft > 0) f.lockLeft--;
if (f.forceSafe > 0) f.forceSafe--;
if (f.allinBlock > 0) f.allinBlock--;
if (f.jpDebuff > 0) f.jpDebuff--;
if (f.php <= 0) return finish(f, false, start);
if (f.bhp <= 0) return finish(f, true, start);
if (mech === "bod" && f.turn >= 18) { ev(f, "mech", 0, { msg: "QUARTERLY GOALS MISSED - contract terminated!" }); return finish(f, false, start); }
if (f.turn >= 60) { ev(f, "mech", 0, { msg: "Overtime - the boss escaped." }); return finish(f, false, start); }
return f.events.slice(start);
}
function bossHit(f, a, src, stance) {
const p = f.p, R = f.R;
if (R() < p.dodge) {
ev(f, "dodge", 0, { src: src });
if (p.cls === "cyber" && p.tier >= 2) { const d = damageBoss(f, a * 0.4); ev(f, "reflect", d); }
if (p.cls === "cyber" && p.tier >= 4 && p.branch === "b") f.ghost = true;
f.hitStreak = 0;
return;
}
let d = a * (0.85 + 0.3 * R());
if (f.b.mech === "broker") d *= 1 + 0.1 * f.weakStacks;
if (stance === "safe" && p.fx.safeDef) d *= 1 - p.fx.safeDef;
damagePlayer(f, d, src);
f.hitStreak++;
if (f.b.mech === "printer" && R() < 0.25) { f.jam = true; ev(f, "mech", 0, { msg: "PAPER JAM - your next attack is blocked!" }); }
if (f.b.mech === "miner") f.jpDebuff = 2;
}
function finish(f, win, start) {
f.over = true; f.win = win;
f.php = Math.max(0, f.php); f.bhp = Math.max(0, f.bhp);
ev(f, win ? "win" : "lose", 0);
return f.events.slice(start);
}
/** Celý souboj s pevnou strategií (pro simulace). policy(f) -> stance */
function simulate(player, boss, seed, policy) {
const f = newFight(player, boss, seed);
while (!f.over) step(f, typeof policy === "function" ? policy(f) : (policy || "normal"));
return f;
}
const PVP_HP = 2.0;                       // v aréně mají oba víc HP, ať duel trvá ~10-16 akcí
function duel(A, B, seed, stA, stB) {
const R = rng(seed || 1);
const mk = (p, st) => {
const hp = Math.round(p.maxHp * PVP_HP);
return { p: p, R: R, stance: st || "normal", hp: hp, max: hp, shield: Math.round(hp * p.shieldPct), ciStacks: 0, prevMult: 0, ghost: false,
jpDebuff: 0, revived: false, dot: 0, sreHalf: false, atkBuff: p.fx.coin ? (R() < 0.5 ? 1.25 : 0.85) : 1 };
};
const S = [mk(A, stA), mk(B, stB)];
const events = [];
const snap = () => [Math.max(0, S[0].hp), Math.max(0, S[1].hp)];
const shs = () => [S[0].shield, S[1].shield];
const push = (e) => { e.hp = snap(); e.sh = shs(); events.push(e); };
const hurt = (side, dmg) => {
const p = side.p;
if (p.cls === "sysadmin" && p.tier >= 4 && p.branch === "a" && side.sreHalf) { dmg *= 0.5; side.sreHalf = false; }
dmg = Math.round(dmg);
const sh = Math.min(side.shield, dmg); side.shield -= sh; side.hp -= dmg - sh;
let revived = false;
if (side.hp <= 0 && p.cls === "sysadmin" && !side.revived) { side.revived = true; side.hp = Math.round(side.max * 0.10); revived = true; }
if (p.cls === "sysadmin" && p.tier >= 4 && p.branch === "a" && dmg > side.max * 0.12) side.sreHalf = true;
return { dmg: dmg, absorbed: sh, revived: revived };
};
let turn = 0, cur = R() < 0.5 ? 0 : 1;
const first = cur;
while (turn < 60) {
turn++;
const a = S[cur], d = S[1 - cur];
if (d.dot > 0) { d.dot--; const x = hurt(d, a.p.atk * 0.4); push({ t: turn, i: cur, type: "dot", val: x.dmg }); if (d.hp <= 0) break; }
const r = rollRow(a, a.stance);
if (r.k === "bf") {
let self = a.p.atk * BACKFIRE_SELF[a.p.cls] * (1 + a.p.fx.bfDmg);
if (a.p.cls === "engineer" && a.p.tier >= 2) self *= 0.7;
if (a.p.cls === "sysadmin" && a.p.tier >= 3) { self *= 0.5; a.shield += Math.round(self); }
const x = hurt(a, self);
push({ t: turn, i: cur, type: "bf", val: x.dmg, rev: x.revived });
a.prevMult = 0; a.ciStacks = 0;
} else if (r.m > 0 && R() < d.p.dodge) {
push({ t: turn, i: cur, type: "dodge" });
if (d.p.cls === "cyber" && d.p.tier >= 2) { const x = hurt(a, a.p.atk * r.m * 0.4); push({ t: turn, i: 1 - cur, type: "reflect", val: x.dmg }); }
if (d.p.cls === "cyber" && d.p.tier >= 4 && d.p.branch === "b") d.ghost = true;
} else {
let atk = a.p.atk * a.atkBuff;
if (a.ghost) { atk *= 1.4; a.ghost = false; }
let dmg = atk * r.m;
if (r.k === "jp" && a.p.cls === "sysadmin" && a.p.tier >= 4 && a.p.branch === "b") dmg += a.shield * 2;
const x = hurt(d, dmg);
push({ t: turn, i: cur, type: "hit", val: x.dmg, kind: r.k, mult: Math.round(r.m * 100) / 100, abs: x.absorbed, rev: x.revived });
if (r.k === "jp" && a.p.cls === "cyber" && a.p.tier >= 3) d.dot = 3;
if (a.p.cls === "sysadmin" && a.p.tier >= 2 && a.stance === "safe" && r.m >= 1) { const h = Math.round(a.max * 0.04); a.hp = Math.min(a.max, a.hp + h); push({ t: turn, i: cur, type: "heal", val: h }); }
if (a.p.cls === "engineer" && a.p.tier >= 3) a.ciStacks = r.m >= 1 ? Math.min(4, a.ciStacks + 1) : 0;
a.prevMult = r.m >= 1 ? r.m : 0;
}
if (S[0].hp <= 0 || S[1].hp <= 0) break;
cur = 1 - cur;
}
let winner;
if (S[0].hp <= 0 && S[1].hp <= 0) winner = 1;              // oba padli (backfire) - vyhrává obránce
else if (S[0].hp <= 0) winner = 1;
else if (S[1].hp <= 0) winner = 0;
else winner = (S[0].hp / S[0].max >= S[1].hp / S[1].max) ? 0 : 1;   // limit tahů: rozhodne % HP
return { win: winner === 0, winner: winner, first: first, turns: turn, events: events, max: [S[0].max, S[1].max], hp: snap() };
}
