function rollVariant() { return { k: Math.floor(Math.random() * MAPS.length), flip: Math.random() < 0.5 }; }
function physStep(r, P) {
r.vy -= G * DT;
const oy = r.y;
r.x += r.vx * DT; r.y += r.vy * DT;
if (r.x < 8) { r.x = 8; r.vx = -r.vx * 0.6; } else if (r.x > W - 8) { r.x = W - 8; r.vx = -r.vx * 0.6; }
for (let k = 0; k < P.length; k++) {
const p = P[k];
if (r.x + 7 < p.x || r.x - 7 > p.x + p.w) continue;
if (r.vy <= 0 && oy >= p.y && r.y <= p.y) { const fall = -r.vy; r.y = p.y; r.vy = 0; r.vx = 0; r.ground = true; return { k: k, fall: fall }; }   // přistání
if (r.vy > 0 && oy + 22 <= p.y - p.h && r.y + 22 > p.y - p.h) { r.y = p.y - p.h - 22; r.vy = 0; r.bonk = 1; }      // hlavou o stůl
}
if (r.y < 24) { const fall = -r.vy; r.y = 24; r.vy = 0; r.vx = 0; r.ground = true; return { k: 0, fall: fall }; }
return null;
}

}
