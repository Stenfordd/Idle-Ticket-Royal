const sc = reward(r.best, r.tier), s = stats();

log(`🧗 CORPORATE CLIMB over: best height ${r.best.toFixed(1)} m → +${sc.toLocaleString()} scrap.`, sc > 0 ? "special" : "warning");

function puff(x, y, n, s) {
for (let i = 0; i < n; i++) jk.fx.push({ x: x + (Math.random() - 0.5) * 10, y: y + 1, vx: (Math.random() - 0.5) * 90 * s, vy: 20 + Math.random() * 50 * s, life: 0.45 + Math.random() * 0.25, max: 0.7 });
}
function step() {
const r = jk.run;
if (r.ground) return;
r.bonk = 0;
const L = physStep(r, jk.plat);
if (L) { r.land = 1; if (L.fall > 250) puff(r.x, r.y, Math.min(14, 4 + Math.round(L.fall / 90)), Math.min(2, L.fall / 600)); }
const m = (r.y - 24) / PXM;
if (m > r.best) r.best = Math.min(TOP_M, m);
}
function loop(now) {
const r = jk.run; if (!r) return;
const el = Math.min(0.1, (now - jk.last) / 1000); jk.last = now;
jk.acc += el;
while (jk.acc >= DT) { step(); jk.acc -= DT; }
jk.fx.forEach(p => { p.x += p.vx * el; p.y += p.vy * el; p.vy -= 160 * el; p.life -= el; });
jk.fx = jk.fx.filter(p => p.life > 0);
if (r.land > 0) r.land = Math.max(0, r.land - el * 5);
if (jk.charging) jk.charge = Math.min(1, (now - jk.cs) / CHARGE_MS);
const left = r.end - now;
const t = $("jk-t"); if (t) t.textContent = "⏱ " + Math.max(0, Math.ceil(left / 1000)) + "s";
const h = $("jk-h"); if (h) h.textContent = ((r.y - 24) / PXM).toFixed(1) + " m (best " + r.best.toFixed(1) + ")";
draw(now);
if ((now | 0) % 6 === 0) info();
if (left <= 0) { finish(); return; }
jk.raf = requestAnimationFrame(loop);
}
