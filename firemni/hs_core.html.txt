function stop() {

function tick() {

const el = document.querySelector(`[data-hold="${key}"]`);
if (!el || el.disabled) { stop(); return; }

function begin(target, x, y) {

if (key && (!el || el.getAttribute("data-hold") !== key)) stop();         // dotyk jinde = konec držení
eatKey = null;
down = true;
if (!el || el.disabled) return;

stop(); key = el.getAttribute("data-hold"); n = 0; fired = false; TT.holdStop = false; startX = x; startY = y;
timer = setTimeout(tick, 400);
}
function release() {
down = false;
if (!key) return;
const wasHold = fired, k = key;
stop();
if (wasHold) { eatKey = k; eatUntil = Date.now() + 450; }
}
