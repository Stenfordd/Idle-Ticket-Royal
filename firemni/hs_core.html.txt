(function holdRepeat() {
let key = null, timer = 0, n = 0, fired = false, down = false, startX = 0, startY = 0, eatKey = null, eatUntil = 0;
function stop() {
if (timer) clearTimeout(timer); timer = 0;
if (key && fired) TT.saveSoon();
TT.holding = false; key = null; fired = false;
}
function tick() {
if (!key) return;
if (!down || TT.holdStop) { TT.holdStop = false; stop(); return; }
const el = document.querySelector(`[data-hold="${key}"]`);
if (!el || el.disabled) { stop(); return; }
fired = true; TT.holding = true; n++;
el.click();
if (TT.holdStop) { TT.holdStop = false; stop(); return; }
timer = setTimeout(tick, Math.max(90, 260 - n * 22));
}
function begin(target, x, y) {
const el = target && target.closest ? target.closest("[data-hold]") : null;
if (key && (!el || el.getAttribute("data-hold") !== key)) stop();         // dotyk jinde = konec držení
eatKey = null;
down = true;
if (!el || el.disabled) return;
if (key === el.getAttribute("data-hold") && !fired) return;             // touchstart + pointerdown téhož dotyku
stop(); key = el.getAttribute("data-hold"); n = 0; fired = false; TT.holdStop = false; startX = x; startY = y;
timer = setTimeout(tick, 400);
}
function release() {
down = false;
if (!key) return;
const wasHold = fired, k = key;
stop();
if (wasHold) { eatKey = k; eatUntil = Date.now() + 450; }
}
document.addEventListener("pointerdown", (e) => { if (e.button > 0) return; begin(e.target, e.clientX, e.clientY); }, true);
document.addEventListener("touchstart", (e) => { const t = e.touches[0]; if (t) begin(e.target, t.clientX, t.clientY); }, { capture: true, passive: true });
const moved = (x, y) => { if (key && !fired && Math.hypot(x - startX, y - startY) > 12) stop(); };   // posouvání stránky neopakuje
document.addEventListener("pointermove", (e) => moved(e.clientX, e.clientY), true);
document.addEventListener("touchmove", (e) => { const t = e.touches[0]; if (t) moved(t.clientX, t.clientY); }, { capture: true, passive: true });
["pointerup", "pointercancel", "touchend", "touchcancel", "dragstart"].forEach(t => document.addEventListener(t, release, true));
window.addEventListener("blur", (e) => { if (e.target === window) release(); });
document.addEventListener("visibilitychange", () => { if (document.hidden) release(); });
document.addEventListener("contextmenu", (e) => { if (key) e.preventDefault(); }, true);
document.addEventListener("click", (e) => {                 // po držení nepustit ještě jeden klik navíc - jen na držené tlačítko

}, true);
})();
