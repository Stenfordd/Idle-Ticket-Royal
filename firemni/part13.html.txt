function spinTo(reels, which, cb) {
const sym = ["🍒", "🔔", "⭐", "7️⃣", "💀"];
which.forEach(i => { const r = $("hs-r" + i); if (r) r.classList.add("spin"); });
const iv = setInterval(() => { which.forEach(i => { const r = $("hs-r" + i); if (r && r.classList.contains("spin")) r.textContent = sym[Math.floor(Math.random() * 5)]; }); }, 70);
which.forEach((i, k) => setTimeout(() => { const r = $("hs-r" + i); if (r) { r.classList.remove("spin"); r.textContent = reels[i]; } }, 380 + k * 240));
setTimeout(() => { clearInterval(iv); cb(); }, 380 + Math.max(0, which.length - 1) * 240 + 200);
}

function endFight() {
if (window.Meta && Meta.onFight) try { Meta.onFight(fight); } catch (e) {}
bossTitle(fight);
if (fightCtx && fightCtx.onEnd) {
const r = fightCtx.onEnd(fight) || {};
result = Object.assign({ win: fight.win, stars: -1, lines: [], m: fightMission, custom: true }, r);
TT.saveSoon(); updateUI(); showResult(); return;
}
const f = fight, m = fightMission, d = HS();
const prev = d.stars[m.id] || 0, prevMask = d.sm[m.id] || 0;
const mask = f.fled ? 0 : H.starMask(f), s = H.starCount(mask);
const newMask = prevMask | mask, total = H.starCount(newMask);
const lines = [];
if (f.win) {
d.wins++;
const first = !prev;
const xp = C.xpFor(m.id, first), gc = Math.round(C.gcFor(m.id, first) * (window.Meta ? Meta.mult("hsgc") : 1));
gameState.coins += gc;
lines.push(`+${fmt(Math.round(xp * (window.Meta ? Meta.mult("hsxp") : 1)))} XP`, `+${fmt(gc)} GC`);
d.sm[m.id] = newMask; d.stars[m.id] = total;
if ((first && TT.rnd() < 0.5) || (total === 3 && prev < 3) || TT.rnd() < 0.12) dropItem(lines);
if (first && (m.kind === "boss" || m.kind === "secret")) {
const slot = C.BOSS_SLOT[m.id] || SLOTS[Math.floor(TT.rnd() * SLOTS.length)];
if (!d.t6[slot]) { d.t6[slot] = true; lines.push(`🔓 Prismatic ${slot} can now drop from chests`); log(`🌌 PRISMATIC UNLOCK: Quantum ${slot} items can now drop from the Company Chest (chest LV 50+).`, "special"); }
if (m.id === 49) TT.announce("BEAT THE BIG MASSI", "HIGH STAKES", 0, gc);
}
gainXp(xp, lines);
if (total > prev) lines.push(`⭐ ${total}/3 stars on this mission`);
log(`🎰 HIGH STAKES: ${escapeHtml(m.name)} cleared ${"★".repeat(s)} - ${lines.join(", ")}`, "special");
} else {
d.losses++;
const xp = f.fled ? 0 : Math.round(C.xpFor(m.id, false) * 0.3);
if (xp) gainXp(xp, lines);
lines.push(xp ? `+${fmt(xp)} XP` : "no reward");
}
result = { win: f.win, stars: s, mask: mask, prevMask: prevMask, prev: prev, lines: lines, m: m };
TT.saveSoon(); updateUI();
showResult();
}
